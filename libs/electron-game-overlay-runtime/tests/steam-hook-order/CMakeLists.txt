cmake_minimum_required(VERSION 3.24)
project(steam_hook_order LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This inert console loader fixture requires Windows/MSVC.")
endif()
set(MINHOOK_SOURCE_DIR "" CACHE PATH "Pinned ReShade deps/minhook checkout")
if(NOT EXISTS "${MINHOOK_SOURCE_DIR}/src/hook.c")
    message(FATAL_ERROR "Supply the existing pinned MINHOOK_SOURCE_DIR; no dependency is fetched.")
endif()
add_library(steam_model SHARED steam_model.cpp
    "${MINHOOK_SOURCE_DIR}/src/hook.c" "${MINHOOK_SOURCE_DIR}/src/buffer.c"
    "${MINHOOK_SOURCE_DIR}/src/trampoline.c" "${MINHOOK_SOURCE_DIR}/src/hde/hde32.c"
    "${MINHOOK_SOURCE_DIR}/src/hde/hde64.c")
target_include_directories(steam_model PRIVATE "${MINHOOK_SOURCE_DIR}/include")
target_compile_definitions(steam_model PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_features(steam_model PRIVATE cxx_std_17)
add_executable(steam_hook_order fixture.cpp)
target_compile_features(steam_hook_order PRIVATE cxx_std_17)
target_compile_definitions(steam_hook_order PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
foreach(target steam_model steam_hook_order)
    target_compile_options(${target} PRIVATE /W4)
    target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()

if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set_target_properties(steam_model PROPERTIES OUTPUT_NAME GameOverlayRenderer64)
else()
    set_target_properties(steam_model PROPERTIES OUTPUT_NAME GameOverlayRenderer)
endif()
