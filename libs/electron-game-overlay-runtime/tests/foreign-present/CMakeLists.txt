cmake_minimum_required(VERSION 3.24)
project(foreign_present LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${RESHAPE_SOURCE}/deps/minhook/src/hook.c")
  message(FATAL_ERROR "Supply the verified ReShade source and MinHook dependency on Windows/MSVC.")
endif()
set(mh "${RESHAPE_SOURCE}/deps/minhook")
set(hook_sources "${mh}/src/hook.c" "${mh}/src/buffer.c" "${mh}/src/trampoline.c" "${mh}/src/hde/hde32.c" "${mh}/src/hde/hde64.c")
add_library(prior_present SHARED renderer.cpp ${hook_sources})
target_compile_definitions(prior_present PRIVATE EGO_PRIOR_RENDERER=1)
set(DETOURS_SOURCE_DIR "" CACHE PATH "Optional pinned official Detours source for an earlier-hook control")
if(DETOURS_SOURCE_DIR)
  file(READ "${CMAKE_CURRENT_SOURCE_DIR}/../early-process-bootstrap/detours-source-lock.json" detours_lock)
  string(JSON dependency_count LENGTH "${detours_lock}" files)
  math(EXPR dependency_last "${dependency_count} - 1")
  foreach(index RANGE 0 ${dependency_last})
    string(JSON relative GET "${detours_lock}" files ${index} path)
    string(JSON expected GET "${detours_lock}" files ${index} lfSha256)
    file(READ "${DETOURS_SOURCE_DIR}/${relative}" body)
    string(REPLACE "\r\n" "\n" body "${body}")
    string(SHA256 actual "${body}")
    if(NOT actual STREQUAL expected)
      message(FATAL_ERROR "Official Detours source identity mismatch: ${relative}")
    endif()
  endforeach()
  add_library(foreign_fixture_detours STATIC
    "${DETOURS_SOURCE_DIR}/src/detours.cpp" "${DETOURS_SOURCE_DIR}/src/modules.cpp"
    "${DETOURS_SOURCE_DIR}/src/disasm.cpp" "${DETOURS_SOURCE_DIR}/src/image.cpp"
    "${DETOURS_SOURCE_DIR}/src/creatwth.cpp" "${DETOURS_SOURCE_DIR}/src/disolx86.cpp"
    "${DETOURS_SOURCE_DIR}/src/disolx64.cpp" "${DETOURS_SOURCE_DIR}/src/disolia64.cpp"
    "${DETOURS_SOURCE_DIR}/src/disolarm.cpp" "${DETOURS_SOURCE_DIR}/src/disolarm64.cpp")
  target_include_directories(foreign_fixture_detours PUBLIC "${DETOURS_SOURCE_DIR}/src")
  target_compile_definitions(foreign_fixture_detours PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  target_compile_options(foreign_fixture_detours PRIVATE /W3)
  add_library(prior_detours SHARED renderer.cpp ${hook_sources})
  target_compile_definitions(prior_detours PRIVATE EGO_PRIOR_RENDERER=1 EGO_PRIOR_DETOURS=1)
  target_link_libraries(prior_detours PRIVATE foreign_fixture_detours)
  configure_file("${DETOURS_SOURCE_DIR}/LICENSE.md" "${CMAKE_CURRENT_BINARY_DIR}/LICENSE.Detours.md" COPYONLY)
endif()
add_library(foreign_renderer SHARED renderer.cpp ${hook_sources})
add_library(owned_adapter SHARED adapter.cpp)
target_include_directories(owned_adapter PRIVATE "${RESHAPE_SOURCE}/include")
add_executable(foreign_present fixture.cpp)
set(fixture_targets prior_present foreign_renderer owned_adapter foreign_present)
if(TARGET prior_detours)
  list(APPEND fixture_targets prior_detours)
endif()
foreach(target IN LISTS fixture_targets)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /EHsc /permissive-)
  target_include_directories(${target} PRIVATE "${mh}/include" "${RESHAPE_SOURCE}/source/ego_owned_late_d3d11")
  target_link_libraries(${target} PRIVATE d3d11 dxgi)
  target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()
