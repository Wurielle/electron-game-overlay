cmake_minimum_required(VERSION 3.24)
project(foreign_present LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${RESHAPE_SOURCE}/deps/minhook/src/hook.c")
  message(FATAL_ERROR "Supply the verified ReShade source and MinHook dependency on Windows/MSVC.")
endif()
set(mh "${RESHAPE_SOURCE}/deps/minhook")
set(hook_sources "${mh}/src/hook.c" "${mh}/src/buffer.c" "${mh}/src/trampoline.c" "${mh}/src/hde/hde32.c" "${mh}/src/hde/hde64.c")
add_library(prior_present SHARED renderer.cpp ${hook_sources})
target_compile_definitions(prior_present PRIVATE EGO_PRIOR_RENDERER=1)
add_library(foreign_renderer SHARED renderer.cpp ${hook_sources})
add_library(owned_adapter SHARED adapter.cpp)
target_include_directories(owned_adapter PRIVATE "${RESHAPE_SOURCE}/include")
add_executable(foreign_present fixture.cpp)
foreach(target prior_present foreign_renderer owned_adapter foreign_present)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /EHsc /permissive-)
  target_include_directories(${target} PRIVATE "${mh}/include" "${RESHAPE_SOURCE}/source/ego_owned_late_d3d11")
  target_link_libraries(${target} PRIVATE d3d11 dxgi)
  target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()
