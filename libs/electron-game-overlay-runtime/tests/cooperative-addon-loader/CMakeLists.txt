cmake_minimum_required(VERSION 3.24)
project(cooperative_addon_loader_fixture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This controlled fixture requires Windows and MSVC.")
endif()
set(ADDON_BUILD_ID "" CACHE STRING "Exact add-on build ID expected by the tested injector")
set(INJECTOR_SOURCE "" CACHE FILEPATH "Exact patched injector source for the read-only session guard unit")
string(LENGTH "${ADDON_BUILD_ID}" ADDON_BUILD_ID_LENGTH)
if(NOT ADDON_BUILD_ID MATCHES "^[0-9A-F]+$" OR NOT ADDON_BUILD_ID_LENGTH EQUAL 32)
    message(FATAL_ERROR "Supply the injector's exact 32-digit ADDON_BUILD_ID.")
endif()
foreach(kind IN ITEMS target host member member_no_cleanup)
    if(kind STREQUAL "target")
        add_executable(loader_fixture_${kind} fixture.cpp)
    else()
        add_library(loader_fixture_${kind} SHARED fixture.cpp)
    endif()
    target_compile_features(loader_fixture_${kind} PRIVATE cxx_std_17)
    if(kind STREQUAL "member_no_cleanup")
        target_compile_definitions(loader_fixture_${kind} PRIVATE FIXTURE_member FIXTURE_no_unload_query)
    endif()
    target_compile_definitions(loader_fixture_${kind} PRIVATE
        UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
        "FIXTURE_${kind}" "ADDON_BUILD_ID=\"${ADDON_BUILD_ID}\"")
endforeach()
if(NOT EXISTS "${INJECTOR_SOURCE}")
    message(FATAL_ERROR "Supply the exact patched INJECTOR_SOURCE used to build the tested injector.")
endif()
file(TO_CMAKE_PATH "${INJECTOR_SOURCE}" INJECTOR_SOURCE_INCLUDE)
add_executable(loader_fixture_session_guard session_guard.cpp)
target_compile_features(loader_fixture_session_guard PRIVATE cxx_std_17)
target_compile_definitions(loader_fixture_session_guard PRIVATE
    UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
    "EGO_INJECTOR_SOURCE=\"${INJECTOR_SOURCE_INCLUDE}\"")
target_link_options(loader_fixture_session_guard PRIVATE /INCREMENTAL:NO)
