cmake_minimum_required(VERSION 3.24)
project(preloaded_injector_loader_race LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This console fixture requires Windows and MSVC.")
endif()
set(INJECTOR_SOURCE_FILE "" CACHE FILEPATH "Exact candidate injector source")
if(NOT EXISTS "${INJECTOR_SOURCE_FILE}")
    message(FATAL_ERROR "Supply INJECTOR_SOURCE_FILE.")
endif()
file(TO_CMAKE_PATH "${INJECTOR_SOURCE_FILE}" INJECTOR_SOURCE_INCLUDE)
get_filename_component(FIXTURE_OWNED_BUILD_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../build" ABSOLUTE)
file(TO_CMAKE_PATH "${FIXTURE_OWNED_BUILD_ROOT}" FIXTURE_OWNED_BUILD_ROOT)
set(ADDON_BUILD_ID "ABCDEF0123456789ABCDEF0123456789" CACHE STRING "Inert fixture build identity")
string(LENGTH "${ADDON_BUILD_ID}" BUILD_ID_LENGTH)
if(NOT ADDON_BUILD_ID MATCHES "^[0-9A-F]+$" OR NOT BUILD_ID_LENGTH EQUAL 32)
    message(FATAL_ERROR "Supply a 32-character uppercase hex ADDON_BUILD_ID.")
endif()
add_library(preload_race_addon SHARED addon.cpp)
add_executable(preload_loader_race fixture.cpp)
foreach(target IN ITEMS preload_race_addon preload_loader_race)
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
        "ADDON_BUILD_ID=\"${ADDON_BUILD_ID}\"")
    target_compile_options(${target} PRIVATE /W4 /EHsc /permissive-)
    target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()
target_compile_definitions(preload_loader_race PRIVATE "EGO_INJECTOR_SOURCE=\"${INJECTOR_SOURCE_INCLUDE}\""
    "EGO_FIXTURE_OWNED_BUILD_ROOT=L\"${FIXTURE_OWNED_BUILD_ROOT}\"")
target_link_libraries(preload_loader_race PRIVATE bcrypt advapi32 psapi)
enable_testing()
add_test(NAME copied_preload_loader_race COMMAND preload_loader_race "$<TARGET_FILE:preload_race_addon>")
set_tests_properties(copied_preload_loader_race PROPERTIES TIMEOUT 40 WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
