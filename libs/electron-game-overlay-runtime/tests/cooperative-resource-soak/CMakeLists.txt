cmake_minimum_required(VERSION 3.24)
project(cooperative_resource_soak LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This controlled fixture requires Windows and MSVC.")
endif()
add_executable(cooperative_resource_soak fixture.cpp)
target_compile_features(cooperative_resource_soak PRIVATE cxx_std_20)
target_compile_definitions(cooperative_resource_soak PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(cooperative_resource_soak PRIVATE ../../src)
target_link_options(cooperative_resource_soak PRIVATE /INCREMENTAL:NO)
enable_testing()
foreach(case IN ITEMS stable-participant lifetime-capacity concurrent-departure admission-transaction)
    add_test(NAME ${case} COMMAND cooperative_resource_soak ${case})
endforeach()
