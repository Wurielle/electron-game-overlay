cmake_minimum_required(VERSION 3.24)
project(rendering_failure_probes LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "These owned console probes require Windows/MSVC.")
endif()
set(RESHADE_INCLUDE_DIRECTORY "" CACHE PATH "Exact production ReShade headers")
foreach(probe IN ITEMS d2d_factory8 on_demand_input repaint_exact_target owned_ego_input)
  add_executable(${probe} ${probe}.cpp)
  target_compile_features(${probe} PRIVATE cxx_std_17)
  target_compile_definitions(${probe} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${probe} PRIVATE /W4 /EHsc /permissive-)
  target_link_options(${probe} PRIVATE /INCREMENTAL:NO)
  target_link_libraries(${probe} PRIVATE d3d11 dxgi d2d1 user32)
endforeach()
target_include_directories(on_demand_input PRIVATE "${RESHADE_INCLUDE_DIRECTORY}")
