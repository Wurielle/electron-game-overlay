cmake_minimum_required(VERSION 3.24)
project(owned_composition_policy LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${RESHAPE_SOURCE}/source/ini_file.cpp")
  message(FATAL_ERROR "Supply the verified production ReShade source on Windows/MSVC.")
endif()
add_executable(owned_policy fixture.cpp "${RESHAPE_SOURCE}/source/ini_file.cpp")
target_compile_features(owned_policy PRIVATE cxx_std_17)
target_compile_definitions(owned_policy PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(owned_policy PRIVATE /W4 /EHsc /permissive- /FI"${CMAKE_CURRENT_SOURCE_DIR}/fixture_pch.hpp")
target_include_directories(owned_policy PRIVATE "${RESHAPE_SOURCE}/source" "${RESHAPE_SOURCE}/deps/utfcpp/source" "${POLICY_GENERATED}")
target_link_libraries(owned_policy PRIVATE user32)
target_link_options(owned_policy PRIVATE /INCREMENTAL:NO)
add_executable(composition_window_inventory window_inventory.cpp)
target_compile_features(composition_window_inventory PRIVATE cxx_std_17)
target_compile_definitions(composition_window_inventory PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(composition_window_inventory PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(composition_window_inventory PRIVATE user32)
target_link_options(composition_window_inventory PRIVATE /INCREMENTAL:NO)
