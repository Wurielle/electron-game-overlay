cmake_minimum_required(VERSION 3.24)
project(owned_config_retirement LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This console fixture requires Windows and MSVC.")
endif()
set(OWNED_CONFIG_HEADER "" CACHE FILEPATH "Exact owned_config_file.hpp under test")
set(RESHADE_SOURCE_DIR "" CACHE PATH "Exact pinned ReShade source root")
if(NOT EXISTS "${OWNED_CONFIG_HEADER}" OR NOT EXISTS "${RESHADE_SOURCE_DIR}/source/ini_file.cpp")
    message(FATAL_ERROR "Supply OWNED_CONFIG_HEADER and RESHADE_SOURCE_DIR.")
endif()
file(TO_CMAKE_PATH "${OWNED_CONFIG_HEADER}" OWNED_CONFIG_HEADER_NORMALIZED)
file(TO_CMAKE_PATH "${RESHADE_SOURCE_DIR}/source/ini_file.cpp" PINNED_INI_CPP)
add_executable(owned_config_retirement fixture.cpp)
target_compile_features(owned_config_retirement PRIVATE cxx_std_17)
target_compile_definitions(owned_config_retirement PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
    EGO_OWNED_CONFIG_HEADER="${OWNED_CONFIG_HEADER_NORMALIZED}" EGO_PINNED_INI_CPP="${PINNED_INI_CPP}")
target_include_directories(owned_config_retirement PRIVATE "${RESHADE_SOURCE_DIR}/source" "${RESHADE_SOURCE_DIR}/deps/utfcpp/source")
target_compile_options(owned_config_retirement PRIVATE /W4 /permissive- /EHsc /utf-8)
target_link_options(owned_config_retirement PRIVATE /INCREMENTAL:NO)
target_link_libraries(owned_config_retirement PRIVATE bcrypt)
