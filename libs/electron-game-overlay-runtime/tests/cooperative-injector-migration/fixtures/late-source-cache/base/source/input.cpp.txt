static constexpr unsigned int raw_input_no_legacy_keyboard = 0x1u;
static constexpr unsigned int raw_input_no_legacy_mouse = 0x2u;
static constexpr unsigned int raw_input_registered_keyboard = 0x4u;
static constexpr unsigned int raw_input_registered_mouse = 0x8u;
static constexpr ULONGLONG raw_input_registration_retry_delay_ms = 1000;

enum class raw_input_registration_query_result
{
	valid,
	retryable_failure,
	unsupported,
};

static raw_input_registration_query_result query_registered_raw_input_devices(std::unordered_map<HWND, unsigned int> &registrations)
{
	constexpr UINT max_registration_count = 1024;
	std::array<RAWINPUTDEVICE, max_registration_count> devices = {};
	UINT device_count = 0;
	UINT required_count = 0;
	SetLastError(ERROR_SUCCESS);
	const UINT probe = GetRegisteredRawInputDevices(nullptr, &required_count, sizeof(RAWINPUTDEVICE));
	if (probe != UINT(-1))
	{
		if (probe == 0 && required_count == 0)
		{
			registrations.clear();
			return raw_input_registration_query_result::valid;
		}
	}
	else if (GetLastError() != ERROR_INSUFFICIENT_BUFFER)
	{
		return raw_input_registration_query_result::retryable_failure;
	}
	if (required_count > max_registration_count)
		return raw_input_registration_query_result::unsupported;

	for (unsigned int attempt = 0; attempt < 4; ++attempt)
	{
		if (required_count == 0)
		{
			registrations.clear();
			return raw_input_registration_query_result::valid;
		}

		UINT capacity = required_count;
		SetLastError(ERROR_SUCCESS);
		const UINT copied = GetRegisteredRawInputDevices(devices.data(), &capacity, sizeof(RAWINPUTDEVICE));
		if (copied != UINT(-1))
		{
			if (copied > devices.size())
				return raw_input_registration_query_result::retryable_failure;
			device_count = copied;
			break;
		}
		if (GetLastError() != ERROR_INSUFFICIENT_BUFFER)
			return raw_input_registration_query_result::retryable_failure;
		if (capacity > max_registration_count)
			return raw_input_registration_query_result::unsupported;
		required_count = capacity;
		if (attempt == 3)
			return raw_input_registration_query_result::retryable_failure;
	}

	registrations.clear();
	bool mouse_seen = false;
	bool keyboard_seen = false;
	for (UINT index = 0; index < device_count; ++index)
	{
		const RAWINPUTDEVICE &device = devices[index];
		if (device.usUsagePage != 1 || (device.usUsage != 0x02 && device.usUsage != 0x06))
			continue;
		if ((device.dwFlags & RIDEV_REMOVE) != 0)
			return raw_input_registration_query_result::unsupported;

		const DWORD exclusive_mode = device.dwFlags & RIDEV_EXMODEMASK;
		if (exclusive_mode == RIDEV_EXCLUDE || exclusive_mode == RIDEV_PAGEONLY)
			continue;
		if (exclusive_mode != 0 && exclusive_mode != RIDEV_NOLEGACY)
			return raw_input_registration_query_result::unsupported;

		const bool keyboard = device.usUsage == 0x06;
		bool &seen = keyboard ? keyboard_seen : mouse_seen;
		if (seen)
			return raw_input_registration_query_result::unsupported;
		seen = true;

		unsigned int &flags = registrations[device.hwndTarget];
		flags |= keyboard ? raw_input_registered_keyboard : raw_input_registered_mouse;
		if (exclusive_mode == RIDEV_NOLEGACY || is_uwp_app())
			flags |= keyboard ? raw_input_no_legacy_keyboard : raw_input_no_legacy_mouse;
	}
	return raw_input_registration_query_result::valid;
}

static bool publish_raw_input_registration_snapshot()
{
	std::unordered_map<HWND, unsigned int> registrations;
	const raw_input_registration_query_result result = query_registered_raw_input_devices(registrations);
	const bool valid = result == raw_input_registration_query_result::valid;
	const bool retryable = result == raw_input_registration_query_result::retryable_failure;
	const ULONGLONG retry_after = retryable ? GetTickCount64() + raw_input_registration_retry_delay_ms : 0;

	const std::unique_lock<std::shared_mutex> lock(s_windows_mutex);
	if (valid)
		s_raw_input_windows.swap(registrations);
	else
		s_raw_input_windows.clear();
	s_raw_input_registration_retry_after.store(retry_after, std::memory_order_release);
	s_raw_input_registration_retryable.store(retryable, std::memory_order_release);
	s_raw_input_registration_authority.store(valid, std::memory_order_release);
	s_raw_input_registrations_seeded.store(true, std::memory_order_release);
	return valid;
}

static bool ensure_raw_input_registrations_seeded(bool wait_for_update = false)
{
	if (s_raw_input_registrations_seeded.load(std::memory_order_acquire))
	{
		if (s_raw_input_registration_authority.load(std::memory_order_acquire))
			return true;
		if (!s_raw_input_registration_retryable.load(std::memory_order_acquire) ||
			GetTickCount64() < s_raw_input_registration_retry_after.load(std::memory_order_acquire))
			return false;
	}

	std::unique_lock<std::mutex> update_lock(s_raw_input_registration_update_mutex, std::defer_lock);
	if (wait_for_update)
		update_lock.lock();
	else if (!update_lock.try_lock())
		return false;

	if (s_raw_input_registrations_seeded.load(std::memory_order_acquire))
	{
		if (s_raw_input_registration_authority.load(std::memory_order_acquire))
			return true;
		if (!s_raw_input_registration_retryable.load(std::memory_order_acquire) ||
			GetTickCount64() < s_raw_input_registration_retry_after.load(std::memory_order_acquire))
			return false;
	}

	const uint64_t previous_generation = s_raw_input_registration_generation.fetch_add(1, std::memory_order_acq_rel);
	assert((previous_generation & 1) == 0);
	const DWORD previous_error = GetLastError();
	const bool valid = publish_raw_input_registration_snapshot();
	s_raw_input_registration_generation.fetch_add(1, std::memory_order_release);
	SetLastError(previous_error);
	return valid;
}

static HWND normalized_root_window(HWND window)
{
	if (window == nullptr)
		return nullptr;
	const HWND root = GetAncestor(window, GA_ROOT);
	return root != nullptr ? root : window;
}

static unsigned int raw_input_registration_flags(
	HWND source_window,
	bool include_focus_following,
	unsigned int &exact_flags,
	unsigned int &focus_following_flags)
{
	exact_flags = 0;
	focus_following_flags = 0;
	const uint64_t generation = s_raw_input_registration_generation.load(std::memory_order_acquire);
	if ((generation & 1) != 0 || !s_raw_input_registration_authority.load(std::memory_order_acquire))
		return 0;

	unsigned int flags = 0;
	if (const auto exact = s_raw_input_windows.find(source_window); exact != s_raw_input_windows.end())
	{
		flags |= exact->second;
		exact_flags = exact->second;
	}

	if (include_focus_following)
	{
		if (const auto focused = s_raw_input_windows.find(nullptr); focused != s_raw_input_windows.end())
		{
			flags |= focused->second;
			focus_following_flags = focused->second;
		}
	}
	if (generation != s_raw_input_registration_generation.load(std::memory_order_acquire))
	{
		exact_flags = 0;
		focus_following_flags = 0;
		return 0;
	}
	return flags;
}

static void add_input_message_flag(reshade::api::input_message &message, reshade::api::input_message_flags flag)
{
	message.flags |= static_cast<uint32_t>(flag);
}

static bool is_foreground_input_window(HWND window)
{
	if (window == nullptr)
		return false;

	const HWND foreground = GetForegroundWindow();
	return foreground == window || foreground == normalized_root_window(window);
}

#if RESHADE_ADDON >= 2
static void invoke_blocked_input_message(reshade::api::input_message &message)
{
	if (!reshade::has_addon_event<reshade::addon_event::input_message>())
		return;

	message.sequence = s_input_message_sequence.fetch_add(1, std::memory_order_relaxed) + 1;
	reshade::invoke_addon_event<reshade::addon_event::input_message>(message);
}
#endif

static bool handle_and_observe_window_message(MSG *message)
{
#if RESHADE_ADDON >= 2
	reshade::api::input_message blocked_message = {};
	reshade::api::input_message *const observation =
		reshade::has_addon_event<reshade::addon_event::input_message>() ? &blocked_message : nullptr;
	const bool blocked = reshade::input::handle_window_message(message, observation);
	if (blocked && observation != nullptr)
		invoke_blocked_input_message(blocked_message);
	return blocked;
#else
	return reshade::input::handle_window_message(message);
#endif
}

reshade::input::input(window_handle window)
	: _window(window)
{
}

std::shared_ptr<reshade::input> reshade::input::register_window(window_handle window)
{
	assert(window != nullptr);

	DWORD process_id = 0;
	GetWindowThreadProcessId(static_cast<HWND>(window), &process_id);
	if (process_id != GetCurrentProcessId())
	{
		reshade::log::message(reshade::log::level::warning, "Cannot capture input for window %p created by a different process (%lu).", window, process_id);
		return nullptr;
	}

	ensure_raw_input_registrations_seeded(true);

	const std::unique_lock<std::shared_mutex> lock(s_windows_mutex);

	const auto insert = s_windows.emplace(static_cast<HWND>(window), std::weak_ptr<input>());

	if (insert.second || insert.first->second.expired())
	{
#if RESHADE_VERBOSE_LOG
		reshade::log::message(reshade::log::level::debug, "Starting input capture for window %p.", window);
#endif

		const auto instance = std::make_shared<input>(window);
		insert.first->second = instance;

		return instance;
	}
	else
	{
		return insert.first->second.lock();
	}
}

static bool get_pointer_mouse_info(const MSG &message, POINTER_INFO &pointer_info)
{
	switch (message.message)
	{
	case WM_POINTERUPDATE:
	case WM_POINTERDOWN:
	case WM_POINTERUP:
	case WM_POINTERENTER:
	case WM_POINTERLEAVE:
	case WM_POINTERWHEEL:
	case WM_POINTERHWHEEL:
		break;
	default:
		return false;
	}

	return GetPointerInfo(GET_POINTERID_WPARAM(message.wParam), &pointer_info) != FALSE &&
		pointer_info.pointerType == PT_MOUSE;
}

bool reshade::input::handle_window_message(const void *message_data, api::input_message *blocked_message)
{
	assert(message_data != nullptr);

	MSG details = *static_cast<const MSG *>(message_data);
	if (blocked_message != nullptr)
	{
		*blocked_message = {};
		blocked_message->source = details.message == WM_INPUT ? api::input_message_source::raw_input : api::input_message_source::window_message;
		blocked_message->source_window = details.hwnd;
		blocked_message->wparam = static_cast<uint64_t>(details.wParam);
		// 'lParam' contains a transient HRAWINPUT for 'WM_INPUT'. The copied raw
		// payload below replaces it, so never publish that handle to an add-on.
		blocked_message->lparam = details.message == WM_INPUT ? 0 : static_cast<int64_t>(details.lParam);
		blocked_message->message = details.message;
		blocked_message->time = details.time;
	}

	POINTER_INFO pointer_info = {};
	const bool is_pointer_mouse_message = get_pointer_mouse_info(details, pointer_info);

	bool is_mouse_message =
		(details.message >= WM_MOUSEFIRST && details.message <= WM_MOUSELAST) ||
		is_pointer_mouse_message;
	bool is_keyboard_message = details.message >= WM_KEYFIRST && details.message <= WM_KEYLAST;
	if (blocked_message != nullptr && details.message != WM_INPUT)
		blocked_message->device_type = is_mouse_message ? api::input_device_type::mouse : api::input_device_type::keyboard;

	// Ignore messages that are not related to mouse or keyboard input
	if (details.message != WM_INPUT && !is_mouse_message && !is_keyboard_message)
		return false;

	ensure_raw_input_registrations_seeded();

	// Guard access to windows list against race conditions
	std::unique_lock<std::shared_mutex> lock(s_windows_mutex);

	// Remove any expired entry from the list
	for (auto it = s_windows.begin(); it != s_windows.end();)
		if (it->second.expired())
			it = s_windows.erase(it);
		else
			++it;

	// Look up the window in the list of known input windows
	auto input_window = s_windows.find(details.hwnd);
	const uint64_t raw_input_registration_generation = s_raw_input_registration_generation.load(std::memory_order_acquire);
	DWORD source_process = 0;
	const bool source_is_current_thread =
		GetWindowThreadProcessId(details.hwnd, &source_process) == GetCurrentThreadId() &&
		source_process == GetCurrentProcessId();
	bool include_focus_following = source_is_current_thread && is_foreground_input_window(details.hwnd);
	if (include_focus_following)
	{
		GUITHREADINFO thread_info = { sizeof(GUITHREADINFO) };
		DWORD focused_process = 0;
		include_focus_following =
			GetGUIThreadInfo(GetCurrentThreadId(), &thread_info) != FALSE &&
			thread_info.hwndFocus != nullptr &&
			GetWindowThreadProcessId(thread_info.hwndFocus, &focused_process) == GetCurrentThreadId() &&
			focused_process == GetCurrentProcessId() &&
			normalized_root_window(thread_info.hwndFocus) == normalized_root_window(details.hwnd);
		if (include_focus_following && details.message == WM_INPUT)
			include_focus_following = GET_RAWINPUT_CODE_WPARAM(details.wParam) == RIM_INPUT &&
				thread_info.hwndFocus == details.hwnd;
	}
	unsigned int exact_raw_input_flags = 0;
	unsigned int focus_following_raw_input_flags = 0;
	const unsigned int raw_input_flags = raw_input_registration_flags(
		details.hwnd,
		include_focus_following,
		exact_raw_input_flags,
		focus_following_raw_input_flags);

	if (input_window == s_windows.end())
	{
		// Walk through the window chain and until an known window is found
		EnumChildWindows(details.hwnd, [](HWND hwnd, LPARAM lparam) -> BOOL {
			auto &input_window = *reinterpret_cast<decltype(s_windows)::iterator *>(lparam);
			// Return true to continue enumeration
			return (input_window = s_windows.find(hwnd)) == s_windows.end();
		}, reinterpret_cast<LPARAM>(&input_window));
	}
	if (input_window == s_windows.end())
	{
		// Some applications handle input in a child window to the main render window
		if (const HWND parent = GetParent(details.hwnd); parent != NULL)
			input_window = s_windows.find(parent);
	}

	if (input_window == s_windows.end() &&
		(raw_input_flags & (raw_input_registered_keyboard | raw_input_registered_mouse)) != 0)
	{
		unsigned int message_registration_flag = is_keyboard_message
			? raw_input_registered_keyboard
			: (is_mouse_message ? raw_input_registered_mouse : 0);
		if (details.message == WM_INPUT)
		{
			RAWINPUTHEADER raw_header = {};
			UINT raw_header_size = sizeof(raw_header);
			if (GetRawInputData(
					reinterpret_cast<HRAWINPUT>(details.lParam),
					RID_HEADER,
					&raw_header,
					&raw_header_size,
					sizeof(raw_header)) != sizeof(raw_header))
				return false;
			message_registration_flag = raw_header.dwType == RIM_TYPEKEYBOARD
				? raw_input_registered_keyboard
				: (raw_header.dwType == RIM_TYPEMOUSE ? raw_input_registered_mouse : 0);
			if (message_registration_flag == 0)
				return false;
		}

		if (message_registration_flag != 0 &&
			(focus_following_raw_input_flags & message_registration_flag) != 0)
		{
			// A focus-following registration belongs only to the focused source
			// window tree. Never project it into an unrelated render root.
			const HWND source_root = normalized_root_window(details.hwnd);
			uint64_t best_frame_count = 0;
			for (auto candidate = s_windows.begin(); candidate != s_windows.end(); ++candidate)
			{
				if (normalized_root_window(candidate->first) != source_root)
					continue;
				const std::shared_ptr<input> candidate_input = candidate->second.lock();
				if (candidate_input == nullptr ||
					(input_window != s_windows.end() && candidate_input->_frame_count <= best_frame_count))
					continue;
				input_window = candidate;
				best_frame_count = candidate_input->_frame_count;
			}
		}
		else if (message_registration_flag != 0 &&
			(exact_raw_input_flags & message_registration_flag) != 0)
		{
			// Explicit raw-input helper HWNDs retain the established process-wide
			// fallback to the window with the most rendering.
			input_window = std::max_element(s_windows.begin(), s_windows.end(),
				[](auto lhs, auto rhs) { return lhs.second.lock()->_frame_count < rhs.second.lock()->_frame_count; });
		}
	}

	if (input_window == s_windows.end())
		return false;

	const std::shared_ptr<input> input = input_window->second.lock();
	// It may happen that the input was destroyed between the removal of expired entries above and here, so need to abort in this case
	if (input == nullptr)
		return false;

	// At this point we have a shared pointer to the input object and no longer reference any memory from the windows list, so can release the lock
	lock.unlock();

	// Calculate window client mouse position
	const bool client_point_valid = ScreenToClient(static_cast<HWND>(input->_window), &details.pt) != FALSE;
	if (blocked_message != nullptr)
	{
		blocked_message->target_window = input->_window;
		blocked_message->client_x = details.pt.x;
		blocked_message->client_y = details.pt.y;
		if (client_point_valid)
			add_input_message_flag(*blocked_message, api::input_message_flags::client_point_valid);
		if (is_foreground_input_window(static_cast<HWND>(input->_window)))
			add_input_message_flag(*blocked_message, api::input_message_flags::foreground);
		// Mouse-in-pointer promotion can coexist with a raw mouse registration.
		// Mark the promoted copy so consumers can give NOLEGACY raw input sole
		// ownership instead of projecting one physical action twice.
		if (is_pointer_mouse_message && (raw_input_flags & raw_input_no_legacy_mouse) != 0)
			add_input_message_flag(*blocked_message, api::input_message_flags::no_legacy);
	}

	// Prevent input threads from modifying input while it is accessed elsewhere
	const std::unique_lock<std::recursive_mutex> input_lock(input->_mutex);

	input->_mouse_position[0] = details.pt.x;
	input->_mouse_position[1] = details.pt.y;

	switch (details.message)
	{
	case WM_INPUT:
		RAWINPUT raw_data;
		if (UINT raw_data_size = sizeof(raw_data);
			GET_RAWINPUT_CODE_WPARAM(details.wParam) != RIM_INPUT || // Ignore all input sink messages (when window is not focused)
			GetRawInputData(reinterpret_cast<HRAWINPUT>(details.lParam), RID_INPUT, &raw_data, &raw_data_size, sizeof(raw_data.header)) == UINT(-1))
			break;
		if (blocked_message != nullptr)
		{
			blocked_message->device_handle = reinterpret_cast<uintptr_t>(raw_data.header.hDevice);
			blocked_message->wparam = static_cast<uint64_t>(raw_data.header.wParam);
			if (GET_RAWINPUT_CODE_WPARAM(raw_data.header.wParam) == RIM_INPUT)
				add_input_message_flag(*blocked_message, api::input_message_flags::foreground);
		}
		switch (raw_data.header.dwType)
		{
		case RIM_TYPEMOUSE:
		{
			is_mouse_message = true;
			if (blocked_message != nullptr)
			{
				blocked_message->device_type = api::input_device_type::mouse;
				blocked_message->mouse.flags = raw_data.data.mouse.usFlags;
				blocked_message->mouse.button_flags = raw_data.data.mouse.usButtonFlags;
				blocked_message->mouse.button_data = raw_data.data.mouse.usButtonData;
				blocked_message->mouse.raw_buttons = raw_data.data.mouse.ulRawButtons;
				blocked_message->mouse.last_x = raw_data.data.mouse.lLastX;
				blocked_message->mouse.last_y = raw_data.data.mouse.lLastY;
				blocked_message->mouse.extra_information = raw_data.data.mouse.ulExtraInformation;
				if ((raw_input_flags & raw_input_no_legacy_mouse) != 0)
					add_input_message_flag(*blocked_message, api::input_message_flags::no_legacy);
			}

			if ((raw_input_flags & raw_input_no_legacy_mouse) == 0)
				break; // Input is already handled (since legacy mouse messages are enabled), so nothing to do here

			const auto has_unowned_raw_transition = [&](unsigned int key, USHORT down_flag, USHORT up_flag) {
				const uint8_t previous = input->_keys[key];
				const bool was_down = (previous & 0x80) != 0;
				const bool owned = (previous & 0x04) != 0;
				return ((raw_data.data.mouse.usButtonFlags & down_flag) != 0 && was_down && !owned) ||
					((raw_data.data.mouse.usButtonFlags & up_flag) != 0 && !owned);
			};
			const bool block_raw_record =
				!has_unowned_raw_transition(VK_LBUTTON, RI_MOUSE_LEFT_BUTTON_DOWN, RI_MOUSE_LEFT_BUTTON_UP) &&
				!has_unowned_raw_transition(VK_RBUTTON, RI_MOUSE_RIGHT_BUTTON_DOWN, RI_MOUSE_RIGHT_BUTTON_UP) &&
				!has_unowned_raw_transition(VK_MBUTTON, RI_MOUSE_MIDDLE_BUTTON_DOWN, RI_MOUSE_MIDDLE_BUTTON_UP) &&
				!has_unowned_raw_transition(VK_XBUTTON1, RI_MOUSE_BUTTON_4_DOWN, RI_MOUSE_BUTTON_4_UP) &&
				!has_unowned_raw_transition(VK_XBUTTON2, RI_MOUSE_BUTTON_5_DOWN, RI_MOUSE_BUTTON_5_UP);
			if (input->is_blocking_mouse_input() && !block_raw_record)
				is_mouse_message = false;

			const auto update_raw_button = [&](unsigned int key, USHORT down_flag, USHORT up_flag) {
				if ((raw_data.data.mouse.usButtonFlags & down_flag) != 0)
					input->_keys[key] = static_cast<uint8_t>(0x88 | (input->is_blocking_mouse_input() && block_raw_record ? 0x04 : 0));
				if ((raw_data.data.mouse.usButtonFlags & up_flag) != 0)
					input->_keys[key] = 0x08;
			};
			update_raw_button(VK_LBUTTON, RI_MOUSE_LEFT_BUTTON_DOWN, RI_MOUSE_LEFT_BUTTON_UP);
			update_raw_button(VK_RBUTTON, RI_MOUSE_RIGHT_BUTTON_DOWN, RI_MOUSE_RIGHT_BUTTON_UP);
			update_raw_button(VK_MBUTTON, RI_MOUSE_MIDDLE_BUTTON_DOWN, RI_MOUSE_MIDDLE_BUTTON_UP);
			update_raw_button(VK_XBUTTON1, RI_MOUSE_BUTTON_4_DOWN, RI_MOUSE_BUTTON_4_UP);
			update_raw_button(VK_XBUTTON2, RI_MOUSE_BUTTON_5_DOWN, RI_MOUSE_BUTTON_5_UP);

			if (raw_data.data.mouse.usButtonFlags & RI_MOUSE_WHEEL)
				input->_mouse_wheel_delta += static_cast<short>(raw_data.data.mouse.usButtonData) / WHEEL_DELTA;
			break;
		}
		case RIM_TYPEKEYBOARD:
			if (blocked_message != nullptr)
			{
				blocked_message->device_type = api::input_device_type::keyboard;
				blocked_message->keyboard.make_code = raw_data.data.keyboard.MakeCode;
				blocked_message->keyboard.flags = raw_data.data.keyboard.Flags;
				blocked_message->keyboard.virtual_key = raw_data.data.keyboard.VKey;
				blocked_message->keyboard.reserved = raw_data.data.keyboard.Reserved;
				blocked_message->keyboard.message = raw_data.data.keyboard.Message;
				blocked_message->keyboard.extra_information = raw_data.data.keyboard.ExtraInformation;
				if ((raw_input_flags & raw_input_no_legacy_keyboard) != 0)
					add_input_message_flag(*blocked_message, api::input_message_flags::no_legacy);
			}
			if (raw_data.data.keyboard.VKey == 0)
				break; // Ignore messages without a valid key code

			is_keyboard_message = true;
			if ((raw_input_flags & raw_input_no_legacy_keyboard) == 0)
				break; // Input is already handled by 'WM_KEYDOWN' and friends (since legacy keyboard messages are enabled), so nothing to do here

			// Filter out prefix messages without a key code
			if (raw_data.data.keyboard.VKey < 0xFF)
			{
				const bool key_down = (raw_data.data.keyboard.Flags & RI_KEY_BREAK) == 0;
				const uint8_t previous = input->_keys[raw_data.data.keyboard.VKey];
				const bool was_down = (previous & 0x80) != 0;
				const bool owned = (previous & 0x04) != 0;
				if (input->is_blocking_keyboard_input() &&
					((key_down && was_down && !owned) || (!key_down && !owned)))
					is_keyboard_message = false;
				input->_keys[raw_data.data.keyboard.VKey] = key_down ? 0x88 : 0x08;
				input->_keys_time[raw_data.data.keyboard.VKey] = details.time;
				if (key_down && input->is_blocking_keyboard_input() && is_keyboard_message)
					input->_keys[raw_data.data.keyboard.VKey] |= 0x04;
			}

			// No 'WM_CHAR' messages are sent if legacy keyboard messages are disabled, so need to generate text input manually here
			// Cannot use the ToUnicode function always as it seems to reset dead key state and thus calling it can break subsequent application input, should be fine here though since the application is already explicitly using raw input
			// Since Windows 10 version 1607 this supports the 0x2 flag, which prevents the keyboard state from being changed, so it is not a problem there anymore either way
			if ((raw_data.data.keyboard.Flags & RI_KEY_BREAK) == 0)
			{
				WCHAR ch[3] = {};
				const int translated = ToUnicode(raw_data.data.keyboard.VKey, raw_data.data.keyboard.MakeCode, input->_keys, ch, 2, 0x2);
				if (translated != 0)
				{
					const unsigned int text_length = static_cast<unsigned int>(std::min(translated < 0 ? -translated : translated, 2));
					input->_text_input += ch;
					if (blocked_message != nullptr)
					{
						blocked_message->text_length = static_cast<uint8_t>(text_length);
						for (unsigned int index = 0; index < text_length; ++index)
							blocked_message->text[index] = static_cast<uint16_t>(ch[index]);
						if (translated < 0)
							add_input_message_flag(*blocked_message, api::input_message_flags::dead_key);
					}
				}
			}
			break;
		}
		break;
	case WM_CHAR:
		input->_text_input += static_cast<wchar_t>(details.wParam);
		[[fallthrough]];
	case WM_SYSCHAR:
		if (blocked_message != nullptr)
		{
			blocked_message->text[0] = static_cast<uint16_t>(details.wParam);
			blocked_message->text_length = 1;
		}
		break;
	case WM_DEADCHAR:
	case WM_SYSDEADCHAR:
		if (blocked_message != nullptr)
		{
			blocked_message->text[0] = static_cast<uint16_t>(details.wParam);
			blocked_message->text_length = 1;
			add_input_message_flag(*blocked_message, api::input_message_flags::dead_key);
		}
		break;
	case WM_UNICHAR:
		if (blocked_message != nullptr && details.wParam != UNICODE_NOCHAR)
		{
			const uint32_t code_point = static_cast<uint32_t>(details.wParam);
			if (code_point <= 0xFFFF)
			{
				blocked_message->text[0] = static_cast<uint16_t>(code_point);
				blocked_message->text_length = 1;
			}
			else if (code_point <= 0x10FFFF)
			{
				const uint32_t surrogate = code_point - 0x10000;
				blocked_message->text[0] = static_cast<uint16_t>(0xD800 + (surrogate >> 10));
				blocked_message->text[1] = static_cast<uint16_t>(0xDC00 + (surrogate & 0x3FF));
				blocked_message->text_length = 2;
			}
		}
		break;
	case WM_KEYDOWN:
	case WM_SYSKEYDOWN:
		assert(details.wParam > 0 && details.wParam < ARRAYSIZE(input->_keys));
		input->_keys[details.wParam] = 0x88;
		input->_keys_time[details.wParam] = details.time;
		if (input->is_blocking_keyboard_input())
			input->_keys[details.wParam] |= 0x04;
		break;
	case WM_KEYUP:
	case WM_SYSKEYUP:
		assert(details.wParam > 0 && details.wParam < ARRAYSIZE(input->_keys));
		// Do not block key up messages if the key down one was not blocked previously (so key does not get stuck for the application)
		if (input->is_blocking_keyboard_input() && (input->_keys[details.wParam] & 0x04) == 0)
			is_keyboard_message = false;
		input->_keys[details.wParam] = 0x08;
		input->_keys_time[details.wParam] = details.time;
		break;
	case WM_LBUTTONDOWN:
	case WM_LBUTTONDBLCLK: // Double clicking generates this sequence: WM_LBUTTONDOWN -> WM_LBUTTONUP -> WM_LBUTTONDBLCLK -> WM_LBUTTONUP, so handle it like a normal down
		input->_keys[VK_LBUTTON] = 0x88;
		break;
	case WM_LBUTTONUP:
		input->_keys[VK_LBUTTON] = 0x08;
		break;
	case WM_RBUTTONDOWN:
	case WM_RBUTTONDBLCLK:
		input->_keys[VK_RBUTTON] = 0x88;
		break;
	case WM_RBUTTONUP:
		input->_keys[VK_RBUTTON] = 0x08;
		break;
	case WM_MBUTTONDOWN:
	case WM_MBUTTONDBLCLK:
		input->_keys[VK_MBUTTON] = 0x88;
		break;
	case WM_MBUTTONUP:
		input->_keys[VK_MBUTTON] = 0x08;
		break;
	case WM_MOUSEWHEEL:
		input->_mouse_wheel_delta += GET_WHEEL_DELTA_WPARAM(details.wParam) / WHEEL_DELTA;
		break;
	case WM_XBUTTONDOWN:
		assert(HIWORD(details.wParam) == XBUTTON1 || HIWORD(details.wParam) == XBUTTON2);
		input->_keys[VK_XBUTTON1 + (HIWORD(details.wParam) - XBUTTON1)] = 0x88;
		break;
	case WM_XBUTTONUP:
		assert(HIWORD(details.wParam) == XBUTTON1 || HIWORD(details.wParam) == XBUTTON2);
		input->_keys[VK_XBUTTON1 + (HIWORD(details.wParam) - XBUTTON1)] = 0x08;
		break;
	case WM_POINTERDOWN:
	case WM_POINTERUP:
		switch (pointer_info.ButtonChangeType)
		{
		case POINTER_CHANGE_FIRSTBUTTON_DOWN:
			input->_keys[VK_LBUTTON] = 0x88;
			break;
		case POINTER_CHANGE_FIRSTBUTTON_UP:
			input->_keys[VK_LBUTTON] = 0x08;
			break;
		case POINTER_CHANGE_SECONDBUTTON_DOWN:
			input->_keys[VK_RBUTTON] = 0x88;
			break;
		case POINTER_CHANGE_SECONDBUTTON_UP:
			input->_keys[VK_RBUTTON] = 0x08;
			break;
		case POINTER_CHANGE_THIRDBUTTON_DOWN:
			input->_keys[VK_MBUTTON] = 0x88;
			break;
		case POINTER_CHANGE_THIRDBUTTON_UP:
			input->_keys[VK_MBUTTON] = 0x08;
			break;
		case POINTER_CHANGE_FOURTHBUTTON_DOWN:
			input->_keys[VK_XBUTTON1] = 0x88;
			break;
		case POINTER_CHANGE_FOURTHBUTTON_UP:
			input->_keys[VK_XBUTTON1] = 0x08;
			break;
		case POINTER_CHANGE_FIFTHBUTTON_DOWN:
			input->_keys[VK_XBUTTON2] = 0x88;
			break;
		case POINTER_CHANGE_FIFTHBUTTON_UP:
			input->_keys[VK_XBUTTON2] = 0x08;
			break;
		default:
			break;
		}
		break;
	case WM_POINTERWHEEL:
		input->_mouse_wheel_delta += GET_WHEEL_DELTA_WPARAM(details.wParam) / WHEEL_DELTA;
		break;
	}

	const bool blocked = (is_mouse_message && input->is_blocking_mouse_input()) ||
		(is_keyboard_message && input->is_blocking_keyboard_input());
	if (details.message == WM_INPUT)
	{
		const unsigned int registered_flag = is_keyboard_message
			? raw_input_registered_keyboard
			: raw_input_registered_mouse;
		if ((raw_input_flags & registered_flag) == 0 ||
			(raw_input_registration_generation & 1) != 0 ||
			raw_input_registration_generation != s_raw_input_registration_generation.load(std::memory_order_acquire))
			return false;
	}
	return blocked;
}

reshade::input::raw_input_buffer_route reshade::input::resolve_raw_input_buffer_route(bool keyboard)
{
	if (!ensure_raw_input_registrations_seeded())
		return {};
	const uint64_t registration_generation = s_raw_input_registration_generation.load(std::memory_order_acquire);
	if ((registration_generation & 1) != 0 ||
		!s_raw_input_registration_authority.load(std::memory_order_acquire))
		return {};

	raw_input_buffer_route route = {};
	route.keyboard = keyboard;
	std::vector<std::pair<HWND, std::shared_ptr<input>>> candidates;

	const unsigned int registered_flag = keyboard ? raw_input_registered_keyboard : raw_input_registered_mouse;
	const unsigned int no_legacy_flag = keyboard ? raw_input_no_legacy_keyboard : raw_input_no_legacy_mouse;
	const DWORD current_thread = GetCurrentThreadId();
	GUITHREADINFO thread_info = { sizeof(GUITHREADINFO) };
	const HWND focused_window = GetGUIThreadInfo(current_thread, &thread_info) != FALSE
		? thread_info.hwndFocus
		: nullptr;
	{
		std::unique_lock<std::shared_mutex> lock(s_windows_mutex);
		if (registration_generation != s_raw_input_registration_generation.load(std::memory_order_acquire) ||
			!s_raw_input_registration_authority.load(std::memory_order_acquire))
			return {};
		for (auto it = s_windows.begin(); it != s_windows.end();)
			if (it->second.expired())
				it = s_windows.erase(it);
			else
				++it;

		const HWND foreground = GetForegroundWindow();
		const HWND foreground_root = foreground != nullptr ? GetAncestor(foreground, GA_ROOT) : nullptr;
		const HWND normalized_foreground_root = foreground_root != nullptr ? foreground_root : foreground;
		for (const auto &[window, flags] : s_raw_input_windows)
		{
			if ((flags & (registered_flag | no_legacy_flag)) != (registered_flag | no_legacy_flag))
				continue;

			const HWND source_window = window != nullptr ? window : focused_window;
			DWORD source_process = 0;
			if (source_window == nullptr || IsWindow(source_window) == FALSE ||
				GetWindowThreadProcessId(source_window, &source_process) != current_thread ||
				source_process != GetCurrentProcessId())
				continue;

			const HWND root = GetAncestor(source_window, GA_ROOT);
			const HWND normalized_root = root != nullptr ? root : source_window;
			if (normalized_foreground_root == nullptr || normalized_root != normalized_foreground_root)
				continue;
			if (route.source_window != nullptr)
				return {}; // Buffered input has no HWND, so ambiguous registrations must fail open.
			route.source_window = source_window;
		}

		if (route.source_window != nullptr)
		{
			const HWND source_root_handle = GetAncestor(static_cast<HWND>(route.source_window), GA_ROOT);
			const HWND source_root = source_root_handle != nullptr ? source_root_handle : static_cast<HWND>(route.source_window);
			for (const auto &[window, weak_input] : s_windows)
			{
				const std::shared_ptr<input> candidate = weak_input.lock();
				if (candidate == nullptr)
					continue;
				const HWND root_handle = GetAncestor(window, GA_ROOT);
				const HWND root = root_handle != nullptr ? root_handle : window;
				if (root == source_root)
					candidates.emplace_back(window, candidate);
			}
			route.registration_generation = registration_generation;
		}
	}

	uint64_t best_frame_count = 0;
	for (const auto &[window, candidate] : candidates)
	{
		const std::unique_lock<std::recursive_mutex> input_lock(candidate->_mutex);
		if (keyboard ? !candidate->_block_keyboard : !candidate->_block_mouse)
			continue;
		if (route.target_input != nullptr && candidate->_frame_count <= best_frame_count)
			continue;
		route.target_input = candidate;
		best_frame_count = candidate->_frame_count;
	}

	if (route.target_input == nullptr)
		return {};
	route.foreground = is_foreground_input_window(static_cast<HWND>(route.target_input->_window));
	POINT cursor = {};
	if (GetCursorPos(&cursor) != FALSE &&
		ScreenToClient(static_cast<HWND>(route.target_input->_window), &cursor) != FALSE)
	{
		route.client_x = cursor.x;
		route.client_y = cursor.y;
		route.client_point_valid = true;
	}
	if (!route.foreground || (!keyboard && !route.client_point_valid))
		return {};
	return route;
}

bool reshade::input::handle_raw_input_buffer(const void *raw_input, const raw_input_buffer_route &route)
{
	if (raw_input == nullptr || !route)
		return false;
	if (route.registration_generation != s_raw_input_registration_generation.load(std::memory_order_acquire))
		return false;

	const RAWINPUT &raw_data = *static_cast<const RAWINPUT *>(raw_input);
	if ((route.keyboard && raw_data.header.dwType != RIM_TYPEKEYBOARD) ||
		(!route.keyboard && raw_data.header.dwType != RIM_TYPEMOUSE) ||
		GET_RAWINPUT_CODE_WPARAM(raw_data.header.wParam) != RIM_INPUT)
		return false;

	api::input_message message = {};
	message.source = api::input_message_source::raw_input_buffer;
	message.device_type = route.keyboard ? api::input_device_type::keyboard : api::input_device_type::mouse;
	message.source_window = route.source_window;
	message.target_window = route.target_input->_window;
	message.device_handle = reinterpret_cast<uintptr_t>(raw_data.header.hDevice);
	message.wparam = static_cast<uint64_t>(raw_data.header.wParam);
	message.message = WM_INPUT;
	message.time = static_cast<uint32_t>(GetMessageTime());
	message.client_x = route.client_x;
	message.client_y = route.client_y;
	add_input_message_flag(message, api::input_message_flags::no_legacy);
	if (route.foreground)
		add_input_message_flag(message, api::input_message_flags::foreground);
	if (route.client_point_valid)
		add_input_message_flag(message, api::input_message_flags::client_point_valid);

	bool block_record = true;
	{
		const std::unique_lock<std::recursive_mutex> input_lock(route.target_input->_mutex);
		if (route.keyboard ? !route.target_input->_block_keyboard : !route.target_input->_block_mouse)
			return false;
		if (route.keyboard)
		{
			message.keyboard.make_code = raw_data.data.keyboard.MakeCode;
			message.keyboard.flags = raw_data.data.keyboard.Flags;
			message.keyboard.virtual_key = raw_data.data.keyboard.VKey;
			message.keyboard.reserved = raw_data.data.keyboard.Reserved;
			message.keyboard.message = raw_data.data.keyboard.Message;
			message.keyboard.extra_information = raw_data.data.keyboard.ExtraInformation;

			const unsigned int virtual_key = raw_data.data.keyboard.VKey;
			const bool down = (raw_data.data.keyboard.Flags & RI_KEY_BREAK) == 0;
			const bool valid_message = down
				? raw_data.data.keyboard.Message == WM_KEYDOWN || raw_data.data.keyboard.Message == WM_SYSKEYDOWN
				: raw_data.data.keyboard.Message == WM_KEYUP || raw_data.data.keyboard.Message == WM_SYSKEYUP;
			if (virtual_key == 0 || virtual_key >= 0xFF || !valid_message)
				return false;

			const uint8_t previous = route.target_input->_keys[virtual_key];
			const bool was_down = (previous & 0x80) != 0;
			const bool owned = (previous & 0x04) != 0;
			if (down)
			{
				block_record = owned || !was_down;
				route.target_input->_keys[virtual_key] = static_cast<uint8_t>(0x88 | (block_record ? 0x04 : 0));
			}
			else
			{
				block_record = owned;
				route.target_input->_keys[virtual_key] = 0x08;
			}
			route.target_input->_keys_time[virtual_key] = message.time;

			if (down && block_record)
			{
				WCHAR text[5] = {};
				const int translated = ToUnicode(
					virtual_key,
					raw_data.data.keyboard.MakeCode,
					route.target_input->_keys,
					text,
					4,
					0x2);
				if (translated != 0)
				{
					const unsigned int text_length = static_cast<unsigned int>(
						std::min(translated < 0 ? -translated : translated, 4));
					route.target_input->_text_input.append(text, text_length);
					message.text_length = static_cast<uint8_t>(text_length);
					for (unsigned int index = 0; index < text_length; ++index)
						message.text[index] = static_cast<uint16_t>(text[index]);
					if (translated < 0)
						add_input_message_flag(message, api::input_message_flags::dead_key);
				}
			}
		}
		else
		{
			message.mouse.flags = raw_data.data.mouse.usFlags;
			message.mouse.button_flags = raw_data.data.mouse.usButtonFlags;
			message.mouse.button_data = raw_data.data.mouse.usButtonData;
			message.mouse.raw_buttons = raw_data.data.mouse.ulRawButtons;
			message.mouse.last_x = raw_data.data.mouse.lLastX;
			message.mouse.last_y = raw_data.data.mouse.lLastY;
			message.mouse.extra_information = raw_data.data.mouse.ulExtraInformation;

			const auto has_unowned_transition = [&](unsigned int key, USHORT down_flag, USHORT up_flag) {
				const uint8_t previous = route.target_input->_keys[key];
				const bool was_down = (previous & 0x80) != 0;
				const bool owned = (previous & 0x04) != 0;
				return ((raw_data.data.mouse.usButtonFlags & down_flag) != 0 && was_down && !owned) ||
					((raw_data.data.mouse.usButtonFlags & up_flag) != 0 && !owned);
			};
			block_record =
				!has_unowned_transition(VK_LBUTTON, RI_MOUSE_LEFT_BUTTON_DOWN, RI_MOUSE_LEFT_BUTTON_UP) &&
				!has_unowned_transition(VK_RBUTTON, RI_MOUSE_RIGHT_BUTTON_DOWN, RI_MOUSE_RIGHT_BUTTON_UP) &&
				!has_unowned_transition(VK_MBUTTON, RI_MOUSE_MIDDLE_BUTTON_DOWN, RI_MOUSE_MIDDLE_BUTTON_UP) &&
				!has_unowned_transition(VK_XBUTTON1, RI_MOUSE_BUTTON_4_DOWN, RI_MOUSE_BUTTON_4_UP) &&
				!has_unowned_transition(VK_XBUTTON2, RI_MOUSE_BUTTON_5_DOWN, RI_MOUSE_BUTTON_5_UP);

			const auto update_button = [&](unsigned int key, USHORT down_flag, USHORT up_flag) {
				const uint8_t previous = route.target_input->_keys[key];
				if ((raw_data.data.mouse.usButtonFlags & down_flag) != 0)
					route.target_input->_keys[key] = static_cast<uint8_t>(0x88 | (block_record ? 0x04 : 0));
				if ((raw_data.data.mouse.usButtonFlags & up_flag) != 0)
					route.target_input->_keys[key] = 0x08;
			};
			update_button(VK_LBUTTON, RI_MOUSE_LEFT_BUTTON_DOWN, RI_MOUSE_LEFT_BUTTON_UP);
			update_button(VK_RBUTTON, RI_MOUSE_RIGHT_BUTTON_DOWN, RI_MOUSE_RIGHT_BUTTON_UP);
			update_button(VK_MBUTTON, RI_MOUSE_MIDDLE_BUTTON_DOWN, RI_MOUSE_MIDDLE_BUTTON_UP);
			update_button(VK_XBUTTON1, RI_MOUSE_BUTTON_4_DOWN, RI_MOUSE_BUTTON_4_UP);
			update_button(VK_XBUTTON2, RI_MOUSE_BUTTON_5_DOWN, RI_MOUSE_BUTTON_5_UP);
			if (block_record && (raw_data.data.mouse.usButtonFlags & RI_MOUSE_WHEEL))
				route.target_input->_mouse_wheel_delta += static_cast<short>(raw_data.data.mouse.usButtonData) / WHEEL_DELTA;
			if (block_record)
			{
				route.target_input->_mouse_position[0] = message.client_x;
				route.target_input->_mouse_position[1] = message.client_y;
			}
		}
	}

	if (!block_record)
		return false;
#if RESHADE_ADDON >= 2
	if (has_addon_event<addon_event::input_message>())
		invoke_blocked_input_message(message);
#endif
	return true;
}

bool reshade::input::is_key_down(unsigned int keycode) const
{
	assert(keycode < ARRAYSIZE(_keys));
	return keycode < ARRAYSIZE(_keys) && (_keys[keycode] & 0x80) == 0x80;
}
bool reshade::input::is_key_pressed(unsigned int keycode) const
{
	assert(keycode < ARRAYSIZE(_keys));
	return keycode > 0 && keycode < ARRAYSIZE(_keys) && (_keys[keycode] & 0x88) == 0x88 && !is_key_repeated(keycode);
}
bool reshade::input::is_key_pressed(unsigned int keycode, bool ctrl, bool shift, bool alt, bool force_modifiers) const
{
	if (keycode == 0)
		return false;

	const bool key_down = is_key_pressed(keycode), ctrl_down = is_key_down(VK_CONTROL), shift_down = is_key_down(VK_SHIFT), alt_down = is_key_down(VK_MENU);
	if (force_modifiers) // Modifier state is required to match
		return key_down && (ctrl == ctrl_down && shift == shift_down && alt == alt_down);
	else // Modifier state is optional and only has to match when down
		return key_down && (!ctrl || ctrl_down) && (!shift || shift_down) && (!alt || alt_down);
}
bool reshade::input::is_key_released(unsigned int keycode) const
{
	assert(keycode < ARRAYSIZE(_keys));
	return keycode > 0 && keycode < ARRAYSIZE(_keys) && (_keys[keycode] & 0x88) == 0x08;
}
bool reshade::input::is_key_repeated(unsigned int keycode) const
{
	assert(keycode < ARRAYSIZE(_keys));
	return keycode < ARRAYSIZE(_keys) && (_last_keys[keycode] & 0x80) == 0x80 && (_keys[keycode] & 0x80) == 0x80;
}

void reshade::input::next_frame()
{
	static const auto GetKeyState_trampoline = reshade::hooks::is_hooked(GetKeyState) ? reshade::hooks::call(HookGetKeyState, GetKeyState) : GetKeyState;
	static const auto GetAsyncKeyState_trampoline = reshade::hooks::is_hooked(GetAsyncKeyState) ? reshade::hooks::call(HookGetAsyncKeyState, GetAsyncKeyState) : GetAsyncKeyState;

	_frame_count++;

	// Backup key states from the last processed frame so that state transitions can be identified
	std::copy_n(_keys, 256, _last_keys);

	for (uint8_t &state : _keys)
		state &= ~0x08;

	// Reset any pressed down key states (apart from mouse buttons) that have not been updated for more than 5 seconds
	// Do not check mouse buttons here, since 'GetAsyncKeyState' always returns the state of the physical mouse buttons, not the logical ones in case they were remapped
	// See https://docs.microsoft.com/windows/win32/api/winuser/nf-winuser-getasynckeystate
	// And time is not tracked for mouse buttons anyway
	const DWORD time = GetTickCount();
	for (unsigned int i = 8; i < 256; ++i)
		if ((_keys[i] & 0x80) != 0 &&
			(time - _keys_time[i]) > 5000 &&
			(GetAsyncKeyState_trampoline(i) & 0x8000) == 0)
			(_keys[i] = 0x08);

	_text_input.clear();
	_mouse_wheel_delta = 0;
	_last_mouse_position[0] = _mouse_position[0];
	_last_mouse_position[1] = _mouse_position[1];

	// Update caps lock state
	_keys[VK_CAPITAL] |= GetKeyState_trampoline(VK_CAPITAL) & 0x1;

	// Update modifier key state
	if ((_keys[VK_MENU] & 0x88) != 0 &&
		(GetKeyState_trampoline(VK_MENU) & 0x8000) == 0)
		(_keys[VK_MENU] = 0x08);

	// Update print screen state (there is no key down message, but the key up one is received via the message queue)
	if ((_keys[VK_SNAPSHOT] & 0x80) == 0 &&
		(GetAsyncKeyState_trampoline(VK_SNAPSHOT) & 0x8000) != 0)
		(_keys[VK_SNAPSHOT] = 0x88),
		(_keys_time[VK_SNAPSHOT] = time);

	// Run through all forms of input blocking for all windows and establish whether any of them are blocking input
	const std::shared_lock<std::shared_mutex> lock(s_windows_mutex);

	s_block_mouse.store(std::any_of(s_windows.cbegin(), s_windows.cend(),
		[](const std::pair<HWND, std::weak_ptr<reshade::input>> &input_window) {
			return !input_window.second.expired() && input_window.second.lock()->is_blocking_mouse_input();
		}));
	s_block_keyboard.store(std::any_of(s_windows.cbegin(), s_windows.cend(),
		[](const std::pair<HWND, std::weak_ptr<reshade::input>> &input_window) {
			return !input_window.second.expired() && input_window.second.lock()->is_blocking_keyboard_input();
		}));
	s_block_cursor_warping.store(std::any_of(s_windows.cbegin(), s_windows.cend(),
		[](const std::pair<HWND, std::weak_ptr<reshade::input>> &input_window) {
			return !input_window.second.expired() && input_window.second.lock()->is_blocking_mouse_cursor_warping();
		}));
}

void reshade::input::block_mouse_input(bool enable)
{
	const std::unique_lock<std::recursive_mutex> input_lock(_mutex);
	if (enable && !_block_mouse)
	{
		static const auto GetAsyncKeyState_trampoline = reshade::hooks::is_hooked(GetAsyncKeyState) ? reshade::hooks::call(HookGetAsyncKeyState, GetAsyncKeyState) : GetAsyncKeyState;
		for (const unsigned int key : { VK_LBUTTON, VK_RBUTTON, VK_MBUTTON, VK_XBUTTON1, VK_XBUTTON2 })
		{
			_keys[key] &= ~0x84;
			if ((GetAsyncKeyState_trampoline(key) & 0x8000) != 0)
				_keys[key] |= 0x80;
		}
	}
	_block_mouse = enable;
}
void reshade::input::block_keyboard_input(bool enable)
{
	const std::unique_lock<std::recursive_mutex> input_lock(_mutex);
	if (enable && !_block_keyboard)
	{
		static const auto GetAsyncKeyState_trampoline = reshade::hooks::is_hooked(GetAsyncKeyState) ? reshade::hooks::call(HookGetAsyncKeyState, GetAsyncKeyState) : GetAsyncKeyState;
		for (unsigned int key = 7; key < ARRAYSIZE(_keys); ++key)
		{
			_keys[key] &= ~0x84;
			if ((GetAsyncKeyState_trampoline(key) & 0x8000) != 0)
				_keys[key] |= 0x80;
		}
	}
	_block_keyboard = enable;
}
void reshade::input::block_mouse_cursor_warping(bool enable)
{
	static const auto ClipCursor_trampoline = reshade::hooks::is_hooked(ClipCursor) ? reshade::hooks::call(HookClipCursor, ClipCursor) : ClipCursor;

	// Some games setup ClipCursor with a tiny area which could make the cursor stay in that area instead of the whole window
	if (enable)
	{
		ClipCursor_trampoline(nullptr);
	}
	else if (_block_cursor_warping && (s_last_clip_cursor.right - s_last_clip_cursor.left) != 0 && (s_last_clip_cursor.bottom - s_last_clip_cursor.top) != 0)
	{
		// Restore previous clipping rectangle when not blocking mouse input
		ClipCursor_trampoline(&s_last_clip_cursor);
	}

	_block_cursor_warping = enable;
}

bool reshade::input::is_blocking_any_mouse_input(window_handle target)
{
	if (target != nullptr)
	{
		const std::shared_lock<std::shared_mutex> lock(s_windows_mutex);

		return std::any_of(s_windows.cbegin(), s_windows.cend(),
			[target](const std::pair<HWND, std::weak_ptr<input>> &input_window) {
				return !input_window.second.expired() && input_window.first == target && input_window.second.lock()->is_blocking_mouse_input();
			});
	}

	return s_block_mouse.load();
}

	if (target != nullptr)

			});
	}

	return s_block_keyboard.load();
}

	return s_block_cursor_warping.load();
}

extern "C" BOOL WINAPI HookGetMessageA(LPMSG lpMsg, HWND hWnd, UINT wMsgFilterMin, UINT wMsgFilterMax)
{
#ifndef RESHADE_TEST_APPLICATION
	DWORD mask = QS_ALLINPUT;
	if (wMsgFilterMin != 0 || wMsgFilterMax != 0)
	{
		mask = QS_POSTMESSAGE | QS_SENDMESSAGE;
		if (wMsgFilterMin <= WM_KEYLAST && wMsgFilterMax >= WM_KEYFIRST)
			mask |= QS_KEY;
		if ((wMsgFilterMin <= WM_MOUSELAST &&
			 wMsgFilterMax >= WM_MOUSEFIRST) ||
			(wMsgFilterMin <= WM_MOUSELAST + WM_NCMOUSEMOVE - WM_MOUSEMOVE &&
			 wMsgFilterMax >= WM_MOUSEFIRST + WM_NCMOUSEMOVE - WM_MOUSEMOVE))
			mask |= QS_MOUSE;
		if (wMsgFilterMin <= WM_TIMER && wMsgFilterMax >= WM_TIMER)
			mask |= QS_TIMER;
		if (wMsgFilterMin <= WM_PAINT && wMsgFilterMax >= WM_PAINT)
			mask |= QS_PAINT;
		if (wMsgFilterMin <= WM_HOTKEY && wMsgFilterMax >= WM_HOTKEY)
			mask |= QS_HOTKEY;
		if (wMsgFilterMin <= WM_INPUT && wMsgFilterMax >= WM_INPUT)
			mask |= QS_RAWINPUT;
	}

	// Implement 'GetMessage' with an additional trigger event (see also DLL_PROCESS_DETACH in dllmain.cpp for more explanation)
	while (!PeekMessageA(lpMsg, hWnd, wMsgFilterMin, wMsgFilterMax, PM_REMOVE))
	{
		assert(g_exit_event != nullptr);

		if (MsgWaitForMultipleObjects(1, &g_exit_event, FALSE, INFINITE, mask) != (WAIT_OBJECT_0 + 1))
		{
			// Clear message structure, so application does not process it
			std::memset(lpMsg, 0, sizeof(MSG));
			// Do not return an error (-1), since this causes WPF to throw an exception
			return 1;
		}
	}
#else
	static const auto trampoline = reshade::hooks::call(HookGetMessageA);
	const BOOL result = trampoline(lpMsg, hWnd, wMsgFilterMin, wMsgFilterMax);
	if (result < 0) // If there is an error, the return value is negative (https://docs.microsoft.com/windows/win32/api/winuser/nf-winuser-getmessage)
		return result;

	assert(lpMsg != nullptr);

	if (lpMsg->hwnd != nullptr && handle_and_observe_window_message(lpMsg))
	{
		// We still want 'WM_CHAR' messages, so translate message
		TranslateMessage(lpMsg);

		// Change message so it is ignored by the recipient window
		lpMsg->message = WM_NULL;
	}
#endif

	return lpMsg->message != WM_QUIT;
}

{

		TranslateMessage(lpMsg);

	}

	return TRUE;

extern "C" BOOL WINAPI HookPostMessageA(HWND hWnd, UINT Msg, WPARAM wParam, LPARAM lParam)
{
	// Do not allow mouse movement simulation while we block input
	if (reshade::input::is_blocking_any_mouse_input(hWnd) && Msg == WM_MOUSEMOVE)
		return TRUE;

	static const auto trampoline = reshade::hooks::call(HookPostMessageA);
	return trampoline(hWnd, Msg, wParam, lParam);
}

extern "C" BOOL WINAPI HookRegisterRawInputDevices(PCRAWINPUTDEVICE pRawInputDevices, UINT uiNumDevices, UINT cbSize)
{
#if RESHADE_VERBOSE_LOG
	reshade::log::message(
		reshade::log::level::debug,
		"Redirecting RegisterRawInputDevices(pRawInputDevices = %p, uiNumDevices = %u, cbSize = %u) ...",
		pRawInputDevices, uiNumDevices, cbSize);
#endif

	for (UINT i = 0; i < uiNumDevices; ++i)
	{
		const RAWINPUTDEVICE &device = pRawInputDevices[i];

#if RESHADE_VERBOSE_LOG
		reshade::log::message(reshade::log::level::debug, "> Dumping device registration at index %u:", i);
		reshade::log::message(reshade::log::level::debug, "  +-----------------------------------------+-----------------------------------------+");
		reshade::log::message(reshade::log::level::debug, "  | Parameter                               | Value                                   |");
		reshade::log::message(reshade::log::level::debug, "  +-----------------------------------------+-----------------------------------------+");
		reshade::log::message(reshade::log::level::debug, "  | UsagePage                               | %-39hu |", device.usUsagePage);
		reshade::log::message(reshade::log::level::debug, "  | Usage                                   | %-39hu |", device.usUsage);
		reshade::log::message(reshade::log::level::debug, "  | Flags                                   | %-39lu |", device.dwFlags);
		reshade::log::message(reshade::log::level::debug, "  | TargetWindow                            | %-39p |", device.hwndTarget);
		reshade::log::message(reshade::log::level::debug, "  +-----------------------------------------+-----------------------------------------+");
#endif

	}

	static const auto trampoline = reshade::hooks::call(HookRegisterRawInputDevices);
	const std::unique_lock<std::mutex> update_lock(s_raw_input_registration_update_mutex);
	const uint64_t previous_generation = s_raw_input_registration_generation.fetch_add(1, std::memory_order_acq_rel);
	assert((previous_generation & 1) == 0);
	const BOOL result = trampoline(pRawInputDevices, uiNumDevices, cbSize);
	const DWORD registration_error = GetLastError();
	publish_raw_input_registration_snapshot();
	s_raw_input_registration_generation.fetch_add(1, std::memory_order_release);

	if (!result)
		reshade::log::message(reshade::log::level::warning, "RegisterRawInputDevices failed with error code %lu.", registration_error);
	SetLastError(registration_error);
	return result;
}

extern "C" BOOL WINAPI HookClipCursor(const RECT *lpRect)
{
	s_last_clip_cursor = (lpRect != nullptr) ? *lpRect : RECT {};

	if (reshade::input::is_blocking_any_mouse_input() || reshade::input::is_blocking_any_mouse_cursor_warping())
		// Some applications clip the mouse cursor, so disable that while we want full control over mouse input
		lpRect = nullptr;

	static const auto trampoline = reshade::hooks::call(HookClipCursor);
	return trampoline(lpRect);
}

extern "C" BOOL WINAPI HookSetCursorPosition(int X, int Y)
{
	s_last_cursor_position.x = X;
	s_last_cursor_position.y = Y;

	if (reshade::input::is_blocking_any_mouse_cursor_warping())
		return TRUE;

	static const auto trampoline = reshade::hooks::call(HookSetCursorPosition);
	return trampoline(X, Y);
}
extern "C" BOOL WINAPI HookGetCursorPosition(LPPOINT lpPoint)
{
	if (reshade::input::is_blocking_any_mouse_input())
	{
		assert(lpPoint != nullptr);

		*lpPoint = s_last_cursor_position;

		return TRUE;
	}

	static const auto trampoline = reshade::hooks::call(HookGetCursorPosition);
	const BOOL result = trampoline(lpPoint);
	if (result)
	{
		assert(lpPoint != nullptr);

		// Update position in case it is not overriden via 'SetCursorPos'
		s_last_cursor_position = *lpPoint;
	}

	return result;
}

extern "C" auto WINAPI HookGetKeyState(int vKey) -> SHORT
{
	// Valid keyboard keys are between 8 and 255
	if ((vKey & 0xF8) != 0)
	{
		if (reshade::input::is_blocking_any_keyboard_input())
			return 0;
	}
	else if (vKey >= VK_LBUTTON && vKey <= VK_XBUTTON2)
	{
		// Some games use this API for mouse buttons
		if (reshade::input::is_blocking_any_mouse_input())
			return 0;
	}

	static const auto trampoline = reshade::hooks::call(HookGetKeyState);
	return trampoline(vKey);
}

extern "C" BOOL WINAPI HookGetKeyboardState(PBYTE lpKeyState)
{
	static const auto trampoline = reshade::hooks::call(HookGetKeyboardState);
	const BOOL result = trampoline(lpKeyState);
	if (result)
	{
		if (reshade::input::is_blocking_any_mouse_input())
			std::memset(lpKeyState, 0, 7);
		if (reshade::input::is_blocking_any_keyboard_input())
			std::memset(lpKeyState + 7, 0, 256 - 7);
	}

	return result;
}

// This variant of raw input does not use 'WM_INPUT' messages, so a hook is necessary to block these inputs
extern "C" UINT WINAPI HookGetRawInputBuffer(PRAWINPUT pData, PUINT pcbSize, UINT cbSizeHeader)
{
	static const auto trampoline = reshade::hooks::call(HookGetRawInputBuffer);
	const UINT result = trampoline(pData, pcbSize, cbSizeHeader);
	const bool block_mouse = reshade::input::is_blocking_any_mouse_input();
	const bool block_keyboard = reshade::input::is_blocking_any_keyboard_input();
	// This is a high throughput API (i.e. 8 kHz mouse polling), so need a fast path to exit
	if (result == static_cast<UINT>(-1) || pData == nullptr || *pcbSize == 0 || !(block_mouse || block_keyboard))
		return result;

	using QWORD = UINT64;
	const reshade::input::raw_input_buffer_route mouse_route = block_mouse
		? reshade::input::resolve_raw_input_buffer_route(false)
		: reshade::input::raw_input_buffer_route {};
	const reshade::input::raw_input_buffer_route keyboard_route = block_keyboard
		? reshade::input::resolve_raw_input_buffer_route(true)
		: reshade::input::raw_input_buffer_route {};

	for (UINT i = 0; i < result; ++i, pData = NEXTRAWINPUTBLOCK(pData))
	{
		switch (pData->header.dwType)
		{
		case RIM_TYPEMOUSE:
			if (reshade::input::handle_raw_input_buffer(pData, mouse_route))
			{
				pData->header.hDevice = nullptr;
				pData->header.wParam = RIM_INPUTSINK;

				// Zero mouse data, to ensure ignore input is ignored even if 'RIM_INPUTSINK' is not respected
				std::memset(&pData->data.mouse, 0, pData->header.dwSize - sizeof(RAWINPUTHEADER));
			}
			break;
		case RIM_TYPEKEYBOARD:
			if (reshade::input::handle_raw_input_buffer(pData, keyboard_route))
			{
				// Supplying an invalid device will early-out SDL before it calls HID APIs to try and get an input report that we don't want it to see
				pData->header.hDevice = nullptr;
				// Most engines will honor the input sink state
				pData->header.wParam = RIM_INPUTSINK;

				if (0 == (pData->data.keyboard.Flags & RI_KEY_BREAK))
				{
					// Fake a key up message
					pData->data.keyboard.Flags |= RI_KEY_BREAK;
					pData->data.keyboard.VKey = 0;
				}
			}
			break;
		}
	}

	return result;
}

#include <concurrent_unordered_map.h>

static concurrency::concurrent_unordered_map<UINT64, std::pair<HHOOK, HOOKPROC>> s_windows_hooks;

template <DWORD hook_id, bool(*is_blocking_input)(reshade::input::window_handle)>
static LRESULT CALLBACK handle_windows_hook(int nCode, WPARAM wParam, LPARAM lParam)
{
	if ((nCode == HC_ACTION || nCode == HC_NOREMOVE) && !is_blocking_input(nullptr))
	{
		const UINT64 global_lookup_key = (static_cast<UINT64>(hook_id) << 32);
		const UINT64 thread_lookup_key = GetCurrentThreadId() | global_lookup_key;

		// Look up original hook function for the current thread and hook type
		if (const auto it = s_windows_hooks.find(thread_lookup_key);
			it != s_windows_hooks.end())
		{
			const HOOKPROC orig_hook_proc = it->second.second;
			// A non-zero result value prevents the system from passing the message to the target window procedure
			// In that case our 'GetMessage' and 'PeekMessage' hooks will not receive them, so ignore that result and continue with the hook chain instead below
			if (orig_hook_proc(nCode, wParam, lParam) == 0)
				return 0;
		}

		// Alternatively look up global hook function for this hook type (registered for thread ID zero)
		if (const auto it = s_windows_hooks.find(global_lookup_key);
			it != s_windows_hooks.end())
		{
			const HOOKPROC orig_hook_proc = it->second.second;
			if (orig_hook_proc(nCode, wParam, lParam) == 0)
				return 0;
		}
	}

	// Bypass the original hook function and continue with the hook chain
	return CallNextHookEx(nullptr, nCode, wParam, lParam);
}

extern "C" HHOOK WINAPI HookSetWindowsHookExA(int idHook, HOOKPROC lpfn, HINSTANCE hmod, DWORD dwThreadId)
{
#if RESHADE_VERBOSE_LOG
	reshade::log::message(reshade::log::level::info, "Redirecting SetWindowsHookExA(idHook = %d, lpfn = %p, hmod = %p, dwThreadId = %lu) ...", idHook, lpfn, hmod, dwThreadId);
#endif

	HOOKPROC orig_hook_proc = lpfn;
	const UINT64 thread_lookup_key = dwThreadId | (static_cast<UINT64>(idHook) << 32);

	if (const auto it = s_windows_hooks.find(thread_lookup_key);
		it == s_windows_hooks.end() || it->second.first == nullptr)
	{
		switch (idHook)
		{
		case WH_MOUSE:
			lpfn = &handle_windows_hook<WH_MOUSE, reshade::input::is_blocking_any_mouse_input>;
			break;
		case WH_MOUSE_LL:
			lpfn = &handle_windows_hook<WH_MOUSE_LL, reshade::input::is_blocking_any_mouse_input>;
			break;
		case WH_KEYBOARD:
			lpfn = &handle_windows_hook<WH_KEYBOARD, reshade::input::is_blocking_any_keyboard_input>;
			break;
		case WH_KEYBOARD_LL:
			lpfn = &handle_windows_hook<WH_KEYBOARD_LL, reshade::input::is_blocking_any_keyboard_input>;
			break;
		}
	}
	else
	{
#if RESHADE_VERBOSE_LOG
		reshade::log::message(reshade::log::level::warning, "Windows hook function of type %d was already registered for thread %lu.", idHook, dwThreadId);
#endif
	}

	static const auto trampoline = reshade::hooks::call(HookSetWindowsHookExA);
	const HHOOK result = trampoline(idHook, lpfn, hmod, dwThreadId);

	if (result != nullptr && lpfn != orig_hook_proc)
	{
#if RESHADE_VERBOSE_LOG
		reshade::log::message(reshade::log::level::info, "Windows hook function of type %d registered for thread %lu.", idHook, dwThreadId);
#endif

		s_windows_hooks[thread_lookup_key] = std::make_pair(result, orig_hook_proc);
	}

	return result;
}

extern "C" BOOL  WINAPI HookUnhookWindowsHookEx(HHOOK hhk)
{
#if RESHADE_VERBOSE_LOG
	reshade::log::message(reshade::log::level::info, "Redirecting UnhookWindowsHookEx(hhk = %p) ...", hhk);
#endif

	for (const std::pair<UINT64, std::pair<HHOOK, HOOKPROC>> &hook : s_windows_hooks)
	{
		if (hook.second.first == hhk)
		{
#if RESHADE_VERBOSE_LOG
			reshade::log::message(reshade::log::level::info, "Windows hook function of type %d unregistered for thread %lu.", static_cast<int>(hook.first >> 32), static_cast<DWORD>(hook.first & 0xFFFFFFFF));
#endif

			s_windows_hooks[hook.first] = std::make_pair(nullptr, nullptr);
			break;
		}
	}

	static const auto trampoline = reshade::hooks::call(HookUnhookWindowsHookEx);
	return trampoline(hhk);
}
