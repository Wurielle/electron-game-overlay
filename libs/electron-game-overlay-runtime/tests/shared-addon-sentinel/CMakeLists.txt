cmake_minimum_required(VERSION 3.24)
project(shared_addon_sentinel LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The ordinary ReShade add-on sentinel requires Windows and MSVC.")
endif()
set(RESHADE_INCLUDE_DIRECTORY "" CACHE PATH "Pinned ReShade include directory")
if(NOT EXISTS "${RESHADE_INCLUDE_DIRECTORY}/reshade.hpp")
    message(FATAL_ERROR "Supply the candidate's RESHADE_INCLUDE_DIRECTORY.")
endif()
foreach(target IN ITEMS shared_addon_sentinel shared_addon_reject)
    add_library(${target} SHARED addon.cpp)
    target_include_directories(${target} PRIVATE "${RESHADE_INCLUDE_DIRECTORY}")
    target_compile_features(${target} PRIVATE cxx_std_17)
    target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_compile_options(${target} PRIVATE /W4 /EHsc /permissive-)
    target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()
target_compile_definitions(shared_addon_reject PRIVATE SHARED_ADDON_REJECT=1)
