cmake_minimum_required(VERSION 3.24)
project(composition_window_eligibility LANGUAGES CXX)
foreach(kind baseline current)
  add_executable(eligibility_${kind} fixture.cpp)
  target_compile_features(eligibility_${kind} PRIVATE cxx_std_17)
  target_compile_definitions(eligibility_${kind} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(kind STREQUAL baseline)
    target_compile_definitions(eligibility_${kind} PRIVATE CURRENT_ELIGIBILITY=0 "EGO_POLICY_HEADER=\"${BASELINE_HEADER}\"")
  else()
    target_compile_definitions(eligibility_${kind} PRIVATE CURRENT_ELIGIBILITY=1 "EGO_POLICY_HEADER=\"${CURRENT_HEADER}\"")
  endif()
  target_compile_options(eligibility_${kind} PRIVATE /W4 /EHsc /permissive-)
  target_link_libraries(eligibility_${kind} PRIVATE user32 comctl32)
  target_link_options(eligibility_${kind} PRIVATE /INCREMENTAL:NO)
endforeach()
