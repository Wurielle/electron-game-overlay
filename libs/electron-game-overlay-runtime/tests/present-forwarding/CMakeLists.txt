cmake_minimum_required(VERSION 3.24)
project(present_forwarding_contract LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The extracted production Present contract requires Windows/MSVC.")
endif()
if(NOT EXISTS "${GENERATED_PRESENT_SOURCE}")
    message(FATAL_ERROR "Run the source-pinned contract generator first.")
endif()
add_executable(present_forwarding_contract fixture.cpp)
target_compile_features(present_forwarding_contract PRIVATE cxx_std_17)
target_compile_definitions(present_forwarding_contract PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN "GENERATED_PRESENT_SOURCE=\"${GENERATED_PRESENT_SOURCE}\"")
target_link_options(present_forwarding_contract PRIVATE /INCREMENTAL:NO)
