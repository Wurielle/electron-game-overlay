cmake_minimum_required(VERSION 3.24)
project(member_diagnostics LANGUAGES CXX)
set(INJECTOR_SOURCE "" CACHE FILEPATH "Pinned production injector")
file(READ "${INJECTOR_SOURCE}" source)
string(REPLACE "\r\n" "\n" source "${source}")
string(SHA256 source_sha "${source}")
if(NOT source_sha STREQUAL EXPECTED_SOURCE_SHA)
  message(FATAL_ERROR "Production source pin mismatch")
endif()
function(extract name first last)
  string(FIND "${source}" "${first}" start)
  string(FIND "${source}" "${last}" end)
  if(start LESS 0 OR end LESS start)
    message(FATAL_ERROR "Missing exact source seam: ${name}")
  endif()
  math(EXPR length "${end}-${start}")
  string(SUBSTRING "${source}" ${start} ${length} body)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${name}.hpp" "${body}")
endfunction()
extract(handles "struct scoped_handle" "struct scoped_mapped_view")
extract(settings "struct target_effective_reshade_settings" "static bool try_resolve_target_environment_directory(")
extract(json "static void append_json_string(" "// Diagnostic reads use")
extract(writer "static bool write_injector_diagnostic(" "static bool write_existing_runtime_load_diagnostic(")
extract(authorization "struct preload_authorization_observation" "static DWORD hash_immutable_native_file(")
add_executable(member_diagnostics fixture.cpp)
target_compile_features(member_diagnostics PRIVATE cxx_std_17)
target_compile_definitions(member_diagnostics PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(member_diagnostics PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(member_diagnostics PRIVATE advapi32)
