cmake_minimum_required(VERSION 3.24)
project(unopened_target_selection LANGUAGES CXX)
set(INJECTOR_SOURCE "" CACHE FILEPATH "Verified production injector source")
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${INJECTOR_SOURCE}")
    message(FATAL_ERROR "Requires Windows/MSVC and production INJECTOR_SOURCE")
endif()
file(READ "${INJECTOR_SOURCE}" injector_text)
string(REPLACE "\r\n" "\n" injector_text "${injector_text}")
string(SHA256 injector_sha "${injector_text}")
if(NOT injector_sha STREQUAL EXPECTED_INJECTOR_SHA256)
    message(FATAL_ERROR "Injector source differs from the supplied production pin")
endif()
add_executable(unopened_target_selection fixture.cpp)
target_compile_features(unopened_target_selection PRIVATE cxx_std_17)
target_compile_definitions(unopened_target_selection PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN "EGO_INJECTOR_SOURCE=\"${INJECTOR_SOURCE}\"")
target_compile_options(unopened_target_selection PRIVATE /W4 /EHsc)
target_link_libraries(unopened_target_selection PRIVATE psapi advapi32)
add_executable(owned_access_refusal_owner owner.cpp)
target_compile_features(owned_access_refusal_owner PRIVATE cxx_std_17)
target_compile_definitions(owned_access_refusal_owner PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(owned_access_refusal_owner PRIVATE advapi32)
add_executable(owned_access_refused_target target.cpp)
