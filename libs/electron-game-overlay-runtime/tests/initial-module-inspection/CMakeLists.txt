cmake_minimum_required(VERSION 3.24)
project(initial_module_inspection_fixture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR (NOT CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 4))
  message(FATAL_ERROR "This deterministic fixture requires Windows MSVC x64 or x86.")
endif()
set(INJECTOR_SOURCE "" CACHE FILEPATH "Exact candidate injector.cpp under test")
if(NOT EXISTS "${INJECTOR_SOURCE}")
  message(FATAL_ERROR "Set INJECTOR_SOURCE to the exact candidate source.")
endif()
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${INJECTOR_SOURCE}")
file(READ "${INJECTOR_SOURCE}" injector_source)
file(SHA256 "${INJECTOR_SOURCE}" injector_raw_sha)
string(REPLACE "\r\n" "\n" injector_source "${injector_source}")
string(SHA256 injector_normalized_sha "${injector_source}")
function(extract_exact output first last)
  string(FIND "${injector_source}" "${first}" start)
  string(FIND "${injector_source}" "${last}" end)
  if(start LESS 0 OR end LESS start)
    message(FATAL_ERROR "Exact source boundary missing: ${first}")
  endif()
  math(EXPR length "${end} - ${start}")
  string(SUBSTRING "${injector_source}" ${start} ${length} body)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}.hpp" "// Exact candidate source extraction.\n${body}")
  string(SHA256 body_sha "${body}")
  set(${output}_sha "${body_sha}" PARENT_SCOPE)
endfunction()
extract_exact(types "enum class module_export_inspection" "static bool read_remote_bytes(")
extract_exact(transient "static bool is_transient_module_inspection_error(" "static target_preflight_result inspect_target_for_reshade(")
extract_exact(preflight "static target_preflight_result inspect_target_for_reshade_once(" "static bool is_transient_module_inspection_error(")
extract_exact(wrapper "static target_preflight_result inspect_target_for_reshade(" "static void append_json_string(")
extract_exact(initial "static DWORD inspect_cooperative_host(" "static DWORD load_participant_only(")
get_filename_component(allowed_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../../build" ABSOLUTE)

target_compile_definitions(initial_module_inspection_fixture PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
  "EGO_ALLOWED_BUILD_ROOT=L\"${allowed_root}\""
  "EGO_SOURCE_RAW_SHA=\"${injector_raw_sha}\""
  "EGO_SOURCE_NORMALIZED_SHA=\"${injector_normalized_sha}\""
  "EGO_TYPES_SHA=\"${types_sha}\"" "EGO_INITIAL_SHA=\"${initial_sha}\""
  "EGO_TRANSIENT_SHA=\"${transient_sha}\"" "EGO_PREFLIGHT_SHA=\"${preflight_sha}\""
  "EGO_WRAPPER_SHA=\"${wrapper_sha}\"")
