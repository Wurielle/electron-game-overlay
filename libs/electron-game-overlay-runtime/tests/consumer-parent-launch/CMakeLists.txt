cmake_minimum_required(VERSION 3.24)
project(consumer_parent_launch LANGUAGES CXX)
# Exercise the actual native launch body from a medium test owner without UAC.
# The only fixture seam forces the existing already-medium early return off.
# This is not evidence of an actual elevated caller and is never packaged.
file(READ "${MANAGER_SOURCE}" source)
string(REPLACE "\r\n" "\n" source "${source}")
string(SHA256 actual "${source}")
if(NOT actual STREQUAL EXPECTED_SOURCE_SHA)
  message(FATAL_ERROR "Production manager source hash mismatch")
endif()
set(seam "const bool supported = !context.elevated && context.integrity <= SECURITY_MANDATORY_MEDIUM_RID;")
string(FIND "${source}" "${seam}" position)
if(position LESS 0)
  message(FATAL_ERROR "Medium early-return seam changed")
endif()
string(LENGTH "${seam}" seam_length)
math(EXPR after_seam "${position}+${seam_length}")
string(SUBSTRING "${source}" ${after_seam} -1 remainder)
string(FIND "${remainder}" "${seam}" duplicate)
if(NOT duplicate EQUAL -1)
  message(FATAL_ERROR "Ambiguous medium early-return seam")
endif()
string(REPLACE "${seam}" "const bool supported = false; // TEST ONLY: exercise real parent launch at medium integrity" source "${source}")
string(REPLACE "resumed = true;" "resumed = true; const DWORD observed_wait = WaitForSingleObject(child_process.get(), 15000); DWORD observed_exit = STILL_ACTIVE; GetExitCodeProcess(child_process.get(), &observed_exit); fprintf(stderr, \"EGO_OWNED_CHILD_EXIT pid=%lu wait=%lu exit=%lu\\n\", child_pid, observed_wait, observed_exit);" source "${source}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fixture.cpp" "${source}")
add_executable(consumer_parent_launch "${CMAKE_CURRENT_BINARY_DIR}/fixture.cpp")
target_compile_features(consumer_parent_launch PRIVATE cxx_std_20)
target_compile_definitions(consumer_parent_launch PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(consumer_parent_launch PRIVATE bcrypt advapi32 userenv)
