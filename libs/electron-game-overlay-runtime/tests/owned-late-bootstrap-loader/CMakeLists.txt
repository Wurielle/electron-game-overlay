cmake_minimum_required(VERSION 3.24)
project(owned_late_bootstrap_loader_fixture LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This inert loader seam fixture requires Windows and MSVC.")
endif()
set(ADDON_BUILD_ID "" CACHE STRING "Exact add-on build ID accepted by the tested injector")
string(LENGTH "${ADDON_BUILD_ID}" ADDON_BUILD_ID_LENGTH)
if(NOT ADDON_BUILD_ID MATCHES "^[0-9A-F]+$" OR NOT ADDON_BUILD_ID_LENGTH EQUAL 32)
    message(FATAL_ERROR "Supply the exact 32-digit ADDON_BUILD_ID of the injector under test.")
endif()
foreach(kind IN ITEMS target owned_host foreign_host malformed_host member)
    if(kind STREQUAL "target")
        add_executable(owned_late_bootstrap_${kind} fixture.cpp)
    else()
        add_library(owned_late_bootstrap_${kind} SHARED fixture.cpp)
    endif()
    target_compile_features(owned_late_bootstrap_${kind} PRIVATE cxx_std_17)
    target_compile_options(owned_late_bootstrap_${kind} PRIVATE /W4 /EHsc /permissive-)
    target_compile_definitions(owned_late_bootstrap_${kind} PRIVATE
        UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN
        "FIXTURE_${kind}" "ADDON_BUILD_ID=\"${ADDON_BUILD_ID}\"")
    target_link_libraries(owned_late_bootstrap_${kind} PRIVATE psapi)
    target_link_options(owned_late_bootstrap_${kind} PRIVATE /INCREMENTAL:NO)
endforeach()
