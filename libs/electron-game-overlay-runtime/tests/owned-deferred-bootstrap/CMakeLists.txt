cmake_minimum_required(VERSION 3.24)
project(owned_deferred_bootstrap LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${RESHAPE_SOURCE}/deps/minhook/src/hook.c")
  message(FATAL_ERROR "Supply the pinned Windows ReShade source cache.")
endif()
set(mh "${RESHAPE_SOURCE}/deps/minhook")
add_library(callback_counter SHARED adapter.cpp)
target_include_directories(callback_counter PRIVATE "${RESHAPE_SOURCE}/include")
add_library(nested_loader SHARED loader.cpp)
add_library(dllmain_loader SHARED loader.cpp)
target_compile_definitions(dllmain_loader PRIVATE LOAD_IN_DLLMAIN=1)
add_executable(owned_deferred_bootstrap fixture.cpp "${mh}/src/hook.c" "${mh}/src/buffer.c" "${mh}/src/trampoline.c" "${mh}/src/hde/hde32.c" "${mh}/src/hde/hde64.c")
target_include_directories(owned_deferred_bootstrap PRIVATE "${mh}/include" "${RESHAPE_SOURCE}/source/ego_owned_late_d3d11")
target_link_libraries(owned_deferred_bootstrap PRIVATE user32)
foreach(target callback_counter nested_loader dllmain_loader owned_deferred_bootstrap)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /EHsc /permissive-)
  target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()
