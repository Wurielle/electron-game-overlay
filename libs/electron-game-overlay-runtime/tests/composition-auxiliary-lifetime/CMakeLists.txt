cmake_minimum_required(VERSION 3.24)
project(composition_auxiliary_lifetime LANGUAGES CXX)
foreach(kind baseline proposed)
  add_executable(lifetime_${kind} fixture.cpp)
  target_compile_features(lifetime_${kind} PRIVATE cxx_std_17)
  target_compile_definitions(lifetime_${kind} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(kind STREQUAL proposed)
    target_compile_definitions(lifetime_${kind} PRIVATE PROPOSED=1 "POLICY_HEADER=\"${CURRENT_HEADER}\"")
  else()
    target_compile_definitions(lifetime_${kind} PRIVATE PROPOSED=0 "POLICY_HEADER=\"${BASELINE_HEADER}\"")
  endif()
  target_compile_options(lifetime_${kind} PRIVATE /W4 /EHsc /permissive-)
  target_link_libraries(lifetime_${kind} PRIVATE user32)
endforeach()
