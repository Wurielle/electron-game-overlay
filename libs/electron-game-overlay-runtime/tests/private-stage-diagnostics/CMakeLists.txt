cmake_minimum_required(VERSION 3.24)
project(private_stage_diagnostics LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The private-stage diagnostic fixture requires Windows/MSVC.")
endif()
file(READ "${INJECTOR_SOURCE}" source)
string(REPLACE "\r\n" "\n" source "${source}")
string(SHA256 actual "${source}")
if(NOT actual STREQUAL "7a6dc34363ff8ebd78484d8aee6ce0463f22e90ab35dd1261fc9fbd9b09a3a3e")
    message(FATAL_ERROR "The exact diagnostic injector source is required.")
endif()
add_executable(private_stage_diagnostics fixture.cpp)
target_compile_features(private_stage_diagnostics PRIVATE cxx_std_17)
target_compile_definitions(private_stage_diagnostics PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN "EGO_INJECTOR_SOURCE=\"${INJECTOR_SOURCE}\"")
target_link_libraries(private_stage_diagnostics PRIVATE psapi advapi32 bcrypt)
target_link_options(private_stage_diagnostics PRIVATE /INCREMENTAL:NO)
