cmake_minimum_required(VERSION 3.24)
project(d3d11_resource_identity LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This console/offscreen fixture requires Windows and MSVC.")
endif()
add_executable(d3d11_resource_identity fixture.cpp)
target_compile_features(d3d11_resource_identity PRIVATE cxx_std_17)
target_compile_definitions(d3d11_resource_identity PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(d3d11_resource_identity PRIVATE /W4 /EHsc /permissive-)
target_link_libraries(d3d11_resource_identity PRIVATE d3d11 dxgi)
target_link_options(d3d11_resource_identity PRIVATE /INCREMENTAL:NO)
