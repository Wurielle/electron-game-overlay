cmake_minimum_required(VERSION 3.24)
project(composition_binding_diagnostics LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The owned observer console fixture requires Windows/MSVC.")
endif()
foreach(kind baseline diagnostic)
  add_executable(binding_${kind} fixture.cpp)
  target_compile_features(binding_${kind} PRIVATE cxx_std_17)
  target_compile_definitions(binding_${kind} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  if(kind STREQUAL baseline)
    target_compile_definitions(binding_${kind} PRIVATE BASELINE_CONTROL=1 "EGO_BINDING_HEADER=\"${BASELINE_HEADER}\"")
  else()
    target_compile_definitions(binding_${kind} PRIVATE BASELINE_CONTROL=0 "EGO_BINDING_HEADER=\"${DIAGNOSTIC_HEADER}\"" "EGO_BINDING_LOGGER=\"${LOGGER_HEADER}\"" "EGO_RESET_HEADER=\"${RESET_HEADER}\"")
  endif()
  target_compile_options(binding_${kind} PRIVATE /W4 /EHsc /permissive-)
  target_link_libraries(binding_${kind} PRIVATE user32)
  target_link_options(binding_${kind} PRIVATE /INCREMENTAL:NO)
endforeach()
