cmake_minimum_required(VERSION 3.24)
project(preload_activation_long_path LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR (NOT CMAKE_SIZEOF_VOID_P EQUAL 8 AND NOT CMAKE_SIZEOF_VOID_P EQUAL 4))
  message(FATAL_ERROR "This isolated console probe requires Windows MSVC x64 or x86.")
endif()
set(OWNED_LATE_SOURCE "" CACHE FILEPATH "Exact owned_late_d3d11.inl source under test")
if(NOT EXISTS "${OWNED_LATE_SOURCE}")
  message(FATAL_ERROR "Set OWNED_LATE_SOURCE to the exact candidate source.")
endif()
file(READ "${OWNED_LATE_SOURCE}" owned_source)
file(SHA256 "${OWNED_LATE_SOURCE}" owned_raw_sha)
string(REPLACE "\r\n" "\n" owned_source "${owned_source}")
string(SHA256 owned_normalized_sha "${owned_source}")
string(FIND "${owned_source}" "    bool ready_word(" ready_start)
string(FIND "${owned_source}" "    void erase_if_current(" ready_end)
if(ready_start LESS 0 OR ready_end LESS ready_start)
  message(FATAL_ERROR "Exact native readiness selector boundaries were not found.")
endif()
math(EXPR ready_length "${ready_end} - ${ready_start}")
string(SUBSTRING "${owned_source}" ${ready_start} ${ready_length} ready_source)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/exact-readiness.hpp" "// Extracted exact candidate functions.\n${ready_source}")
string(FIND "${ready_source}" "K32EnumProcessModules" k32_selector)
set(selector_uses_k32 0)
if(NOT k32_selector LESS 0)
  set(selector_uses_k32 1)
endif()
get_filename_component(allowed_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../../build" ABSOLUTE)
file(TO_CMAKE_PATH "${allowed_root}" allowed_root)
add_library(preload_activation_member SHARED member.cpp)
add_library(preload_activation_member_legacy SHARED member.cpp)
add_library(preload_activation_member_invalid SHARED member.cpp)
add_library(preload_activation_member_unrelated SHARED member.cpp)
target_compile_definitions(preload_activation_member_legacy PRIVATE EGO_FIXTURE_LEGACY)
target_compile_definitions(preload_activation_member_invalid PRIVATE EGO_FIXTURE_INVALID_ABI)
target_compile_definitions(preload_activation_member_unrelated PRIVATE EGO_FIXTURE_NO_ABI)
add_executable(preload_activation_fixture fixture.cpp)
foreach(target IN ITEMS preload_activation_member preload_activation_member_legacy preload_activation_member_invalid preload_activation_member_unrelated preload_activation_fixture)
  target_compile_features(${target} PRIVATE cxx_std_17)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(${target} PRIVATE /W4 /EHsc /permissive-)
  target_link_options(${target} PRIVATE /INCREMENTAL:NO)
endforeach()
target_include_directories(preload_activation_fixture PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_definitions(preload_activation_fixture PRIVATE
  "EGO_ALLOWED_TEST_ROOT=L\"${allowed_root}\""
  "EGO_SOURCE_RAW_SHA=\"${owned_raw_sha}\""
  "EGO_SOURCE_NORMALIZED_SHA=\"${owned_normalized_sha}\""
  "EGO_SELECTOR_USES_K32=${selector_uses_k32}")
target_link_libraries(preload_activation_fixture PRIVATE psapi)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fixture.manifest" [=[<?xml version="1.0" encoding="UTF-8" standalone="yes"?>
<assembly xmlns="urn:schemas-microsoft-com:asm.v1" manifestVersion="1.0"><application xmlns="urn:schemas-microsoft-com:asm.v3"><windowsSettings><longPathAware xmlns="http://schemas.microsoft.com/SMI/2016/WindowsSettings">false</longPathAware></windowsSettings></application></assembly>
]=])
target_link_options(preload_activation_fixture PRIVATE /MANIFEST "/MANIFESTINPUT:${CMAKE_CURRENT_BINARY_DIR}/fixture.manifest")
