cmake_minimum_required(VERSION 3.24)
project(initial_module_inspection_console LANGUAGES CXX)
set(INJECTOR_SOURCE "" CACHE FILEPATH "Verified reconstructed production source")
if(NOT WIN32 OR NOT MSVC OR NOT EXISTS "${INJECTOR_SOURCE}")
    message(FATAL_ERROR "Requires Windows/MSVC and INJECTOR_SOURCE")
endif()
add_executable(initial_module_inspection fixture.cpp)
target_compile_features(initial_module_inspection PRIVATE cxx_std_17)
target_compile_definitions(initial_module_inspection PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN "EGO_INJECTOR_SOURCE=\"${INJECTOR_SOURCE}\"")
target_compile_options(initial_module_inspection PRIVATE /W4 /EHsc)
target_link_libraries(initial_module_inspection PRIVATE psapi advapi32)
add_executable(startup_readiness startup-readiness.cpp)
target_compile_features(startup_readiness PRIVATE cxx_std_17)
target_compile_definitions(startup_readiness PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN "EGO_INJECTOR_SOURCE=\"${INJECTOR_SOURCE}\"")
target_link_libraries(startup_readiness PRIVATE psapi advapi32)
file(READ "${INJECTOR_SOURCE}" readiness_source)
foreach(part IN ITEMS types wait)
    if(part STREQUAL "types")
        set(first "enum class initial_loader_state")
        set(last "// Caller has already validated architecture")
    else()
        set(first "static DWORD wait_for_initial_loader(")
        set(last "static bool write_initial_loader_readiness(")
    endif()
    string(FIND "${readiness_source}" "${first}" start)
    string(FIND "${readiness_source}" "${last}" end)
    if(start LESS 0 OR end LESS start)
        message(FATAL_ERROR "Missing pinned startup readiness seam")
    endif()
    math(EXPR length "${end}-${start}")
    string(SUBSTRING "${readiness_source}" ${start} ${length} body)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/readiness-${part}.hpp" "${body}")
endforeach()
add_executable(startup_readiness_seams startup-readiness-seams.cpp)
target_compile_features(startup_readiness_seams PRIVATE cxx_std_17)
target_compile_definitions(startup_readiness_seams PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(startup_readiness_seams PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
add_library(private_image_marker SHARED private-image-marker.cpp)
add_executable(private_image_identity private-image-identity.cpp)
target_compile_features(private_image_identity PRIVATE cxx_std_17)
target_compile_definitions(private_image_identity PRIVATE UNICODE _UNICODE)
