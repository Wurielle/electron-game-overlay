cmake_minimum_required(VERSION 3.24)
project(reshade_addon_isolation_probe LANGUAGES CXX)

if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This research probe requires Windows x64 and MSVC.")
endif()
set(RESHADE_INCLUDE_DIRECTORY "" CACHE PATH "Unmodified ReShade 6.7.3 public API-18 headers")
if(NOT EXISTS "${RESHADE_INCLUDE_DIRECTORY}/reshade.hpp")
    message(FATAL_ERROR "Set RESHADE_INCLUDE_DIRECTORY to the official API-18 include directory.")
endif()

add_executable(reshade_addon_isolation_host host.cpp)
target_link_libraries(reshade_addon_isolation_host PRIVATE user32 d3d11)
target_compile_features(reshade_addon_isolation_host PRIVATE cxx_std_17)

foreach(probe IN ITEMS a b duplicate)
    add_library(probe_${probe} SHARED addon.cpp)
    target_include_directories(probe_${probe} PRIVATE "${RESHADE_INCLUDE_DIRECTORY}")
    target_compile_features(probe_${probe} PRIVATE cxx_std_17)
    if(probe STREQUAL "b")
        target_compile_definitions(probe_${probe} PRIVATE PROBE_NAME="EGO isolation probe B v2")
    else()
        target_compile_definitions(probe_${probe} PRIVATE PROBE_NAME="EGO isolation probe A v1")
    endif()
endforeach()
