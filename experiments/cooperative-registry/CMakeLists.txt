cmake_minimum_required(VERSION 3.24)
project(ego_cooperative_registry_tests LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)

endif()
add_executable(cooperative_registry_tests tests.cpp)
target_compile_features(cooperative_registry_tests PRIVATE cxx_std_20)
target_include_directories(cooperative_registry_tests PRIVATE ../../libs/electron-game-overlay-runtime/src)
enable_testing()
add_test(NAME cooperative_registry COMMAND cooperative_registry_tests)
